@Override
public void forceCleanupUserSessions(Long userId) {
    if (userId == null) return;
    
    int cleanedCount = 0;
    Iterator<Map.Entry<String, SessionInfo>> iterator = activeSessions.entrySet().iterator();
    
    while (iterator.hasNext()) {
        Map.Entry<String, SessionInfo> entry = iterator.next();
        SessionInfo sessionInfo = entry.getValue();
        
        if (sessionInfo != null && userId.equals(sessionInfo.getUserId())) {
            iterator.remove();
            interruptFlags.remove(entry.getKey());
            cleanedCount++;
            log.info("强制清理用户{}的会话: {}", userId, entry.getKey());
        }
    }
    
    log.info("用户{}会话强制清理完成，清理数量: {}", userId, cleanedCount);
}

### 分布式锁降级方案详细设计

#### 降级触发条件
```java
// 在RedisPaymentSecurityService中实现智能降级
private static final int MAX_REDIS_RETRY = 3;
private static final long REDIS_TIMEOUT_MS = 1000;

public boolean acquirePaymentSession(Long userId) {
    // 1. 优先尝试Redis锁
    for (int i = 0; i < MAX_REDIS_RETRY; i++) {
        try {
            Boolean acquired = redisTemplate.opsForValue()
                .setIfAbsent(getPaymentSessionKey(userId), "locked", Duration.ofMinutes(5));
            if (acquired != null && acquired) {
                log.debug("Redis支付会话锁获取成功，用户ID: {}, 重试次数: {}", userId, i);
                return true;
            }
        } catch (Exception e) {
            log.warn("Redis支付会话锁获取异常，用户ID: {}, 重试次数: {}/{}", userId, i + 1, MAX_REDIS_RETRY, e);
            if (i == MAX_REDIS_RETRY - 1) {
                // 最后一次重试失败，启用降级方案
                return acquirePaymentSessionFromDatabase(userId);
            }
            // 短暂等待后重试
            try { Thread.sleep(100); } catch (InterruptedException ie) { Thread.currentThread().interrupt(); }
        }
    }
    return false;
}
```

#### 数据库降级实现
```java
private boolean acquirePaymentSessionFromDatabase(Long userId) {
    String lockKey = "payment:db:" + userId;
    log.info("启用数据库降级方案获取支付会话锁，用户ID: {}", userId);

    Boolean result = distributedLockService.executeWithLock(lockKey, 3, 300, () -> {
        // 在数据库中记录会话锁状态
        try {
            int inserted = paymentSessionMapper.insertSessionLock(userId, LocalDateTime.now().plusMinutes(5));
            return inserted > 0;
        } catch (DuplicateKeyException e) {
            // 会话锁已存在
            log.debug("数据库支付会话锁已存在，用户ID: {}", userId);
            return false;
        }
    });

    return result != null && result;
}
```

### 监控告警系统详细设计

#### 支付会话锁监控
```java
@Component
public class PaymentSessionMonitor {

    private final RedisTemplate<String, String> redisTemplate;
    private final AlertService alertService;

    @Scheduled(fixedRate = 60000) // 每分钟执行
    public void monitorPaymentSessions() {
        try {
            Set<String> lockKeys = redisTemplate.keys("payment:session:*");
            Map<String, Long> longRunningLocks = new HashMap<>();

            for (String key : lockKeys) {
                Long ttl = redisTemplate.getExpire(key, TimeUnit.SECONDS);
                if (ttl != null && ttl > 0) {
                    // 计算锁已持有时间
                    long heldTime = 300 - ttl; // 5分钟 - 剩余时间
                    if (heldTime > 240) { // 超过4分钟的锁
                        String userId = key.replace("payment:session:", "");
                        longRunningLocks.put(userId, heldTime);
                    }
                }
            }

            if (!longRunningLocks.isEmpty()) {
                alertService.sendPaymentLockAlert(longRunningLocks);
            }

        } catch (Exception e) {
            log.error("支付会话锁监控异常", e);
        }
    }
}
```

@Component
public class ChatSessionMonitor {

    @Scheduled(fixedRate = 300000) // 每5分钟执行
    public void monitorChatSessions() {
        try {
            Map<String, Object> stats = chatInterruptService.getSessionStatistics();

            int activeCount = (Integer) stats.get("activeSessionCount");
            int timeoutCount = (Integer) stats.get("timeoutSessionCount");
            List<String> longRunningSessions = (List<String>) stats.get("longRunningSessions");

            // 检查异常情况
            if (activeCount > 1000) { // 活跃会话过多
                alertService.sendChatSessionAlert("活跃会话数量异常", stats);
            }

            if (timeoutCount > 100) { // 超时会话过多
                alertService.sendChatSessionAlert("超时会话数量异常", stats);
            }

            if (!longRunningSessions.isEmpty()) { // 长时间运行的会话
                alertService.sendChatSessionAlert("长时间运行会话检测", stats);
            }

        } catch (Exception e) {
            log.error("聊天会话监控异常", e);
        }
    }
}
