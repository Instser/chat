```java
/**
 * 临时测试接口：激活用户套餐
 */
@PostMapping("/test/activate-package")
public Result<String> activatePackage(@RequestBody Map<String, Object> request) {
    try {
        Long userId = Long.valueOf(request.get("userId").toString());
        Long packageId = Long.valueOf(request.get("packageId").toString());
        Integer validityDays = Integer.valueOf(request.get("validityDays").toString());
        Integer totalChatCount = Integer.valueOf(request.get("totalChatCount").toString());
        
        boolean success = userAccountService.activatePackage(userId, packageId, validityDays, totalChatCount);
        
        if (success) {
            return Result.success("套餐激活成功", "用户" + userId + "的套餐已激活");
        } else {
            return Result.failed("套餐激活失败");
        }
    } catch (Exception e) {
        log.error("激活套餐异常", e);
        return Result.failed("激活套餐异常：" + e.getMessage());
    }
}
```
