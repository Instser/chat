## 一、实施目标

### 1.1 核心目标
- 实现三级缓存架构（L1: Caffeine + L2: Redis + L3: MySQL）

### 1.2 性能指标

- 权限验证响应时间：减少 50% 以上

## 二、技术架构设计

### 2.1 三级缓存架构
```
L1: Caffeine本地缓存 (5-15分钟)
├── 用户账户信息缓存 (15分钟)
├── 模型权限缓存 (30分钟)
└── 套餐配置缓存 (1小时)

L2: Redis分布式缓存 (15-60分钟)
├── 每日使用统计 (1小时)
├── 用户基本信息 (30分钟)
└── 系统配置缓存 (2小时)

L3: MySQL数据库 (兜底数据源)
└── 持久化数据存储
```

### 2.2 缓存键命名规范
```
格式: 业务模块:数据类型:标识符[:附加参数]

示例:
- user:account:12345 (用户账户)
- package:config:1 (套餐配置)
- model:permission:12345:1 (用户模型权限)
- daily:usage:12345:claude-3 (每日使用统计)
- session:info:12345:session-abc (会话信息)
```

```java
@Component
@Slf4j
public class HybridCacheManager {
    
    @Autowired
    private CacheManager caffeineCacheManager;
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    /**
     * 获取缓存数据，支持三级缓存
     */
    public <T> T get(String key, Supplier<T> supplier, Duration ttl) {
        // L1: Caffeine本地缓存查询
        T result = getFromCaffeine(key);
        if (result != null) {
            return result;
        }
        
        // L2: Redis分布式缓存查询
        result = getFromRedis(key);
        if (result != null) {
            putToCaffeine(key, result, ttl);
            return result;
        }
        
        // L3: 数据源查询
        result = supplier.get();
        if (result != null) {
            putToRedis(key, result, ttl);
            putToCaffeine(key, result, ttl);
        }
        
        return result;
    }

}
```

#### 3.2.1 修改 UserAccountServiceImpl
```java
@Service
@Slf4j
public class UserAccountServiceImpl extends ServiceImpl<UserAccountDao, UserAccount>
        implements UserAccountService {

    @Autowired
    private HybridCacheManager hybridCacheManager;

    private static final String USER_ACCOUNT_CACHE_PREFIX = "user:account:";
    private static final Duration USER_ACCOUNT_TTL = Duration.ofMinutes(15);

    @Override
    public UserAccount getUserAccount(Long userId) {
        if (userId == null) {
            return null;
        }
        
        String cacheKey = USER_ACCOUNT_CACHE_PREFIX + userId;
        return hybridCacheManager.get(cacheKey, () -> {
            UserAccount account = baseMapper.selectByUserId(userId);
            if (account != null) {
                LogUtils.logUserDebug(log, userId, "缓存未命中", "从数据库加载用户账户信息");
            }
            return account;
        }, USER_ACCOUNT_TTL);
    }

    @Override
    public boolean updateUsage(Long userId, Integer usedCount) {
        try {
            // 执行数据库更新
            boolean result = performDatabaseUpdate(userId, usedCount);
            
            if (result) {
                // 更新成功后清除相关缓存
                evictUserAccountCache(userId);
                LogUtils.logUserDebug(log, userId, "更新使用次数", "使用次数更新成功，已清除缓存");
            }
            
            return result;
        } catch (Exception e) {
            LogUtils.logUserError(log, userId, "更新使用次数", "更新使用次数异常", e);
            return false;
        }
    }

    private void evictUserAccountCache(Long userId) {
        String cacheKey = USER_ACCOUNT_CACHE_PREFIX + userId;
        hybridCacheManager.evict(cacheKey);
    }
    
    // 其他方法实现...
}
```

#### 3.2.2 添加缓存预热机制

```java
@Component
@Slf4j
public class UserAccountCacheWarmup {
    
    @Autowired
    private UserAccountService userAccountService;
    
    @EventListener(ApplicationReadyEvent.class)
    public void warmupUserAccountCache() {
        log.info("开始预热用户账户缓存...");
        
        try {
            // 预热活跃用户的账户信息
            List<Long> activeUserIds = userAccountService.getActiveUserIds(1000);
            
            activeUserIds.parallelStream()
                .forEach(userId -> {
                    try {
                        userAccountService.getUserAccount(userId);
                    } catch (Exception e) {
                        log.warn("预热用户{}账户缓存失败: {}", userId, e.getMessage());
                    }
                });
                
            log.info("用户账户缓存预热完成，预热用户数: {}", activeUserIds.size());
        } catch (Exception e) {
            log.error("用户账户缓存预热异常", e);
        }
    }
}
```

### 3.3 第三步：实现 ChatLimitServiceImpl 缓存（2-3天）

#### 3.3.1 修改 ChatLimitServiceImpl
```java
@Service
@Slf4j
public class ChatLimitServiceImpl implements ChatLimitService {

    @Autowired
    private HybridCacheManager hybridCacheManager;
    
    @Autowired
    private UserAccountService userAccountService;
    
    @Autowired
    private PackageService packageService;

    private static final String MODEL_PERMISSION_CACHE_PREFIX = "model:permission:";
    private static final String PACKAGE_CONFIG_CACHE_PREFIX = "package:config:";
    private static final String DAILY_USAGE_CACHE_PREFIX = "daily:usage:";

    @Override
    public String validateChatPermission(Long userId, String sessionId, Long modelId) {
        try {
            // 1. 缓存用户账户信息
            UserAccount userAccount = getCachedUserAccount(userId);
            if (userAccount == null) {
                return "PACKAGE_EXPIRED";
            }

            // 2. 缓存套餐配置
            Package packageConfig = getCachedPackageConfig(userAccount.getPackageId());
            if (packageConfig == null) {
                return "PACKAGE_EXPIRED";
            }

            // 3. 缓存模型权限验证
            String modelPermissionResult = getCachedModelPermission(userId, modelId, packageConfig);
            if (modelPermissionResult != null) {
                return modelPermissionResult;
            }

            // 4. 缓存每日使用统计验证
            String dailyUsageResult = getCachedDailyUsageValidation(userId, modelId);
            if (dailyUsageResult != null) {
                return dailyUsageResult;
            }

            return null; // 验证通过
        } catch (Exception e) {
            LogUtils.logUserError(log, userId, "权限验证", "权限验证异常", e);
            return "SYSTEM_ERROR";
        }
    }

    private UserAccount getCachedUserAccount(Long userId) {
        return userAccountService.getUserAccount(userId); // 已实现缓存
    }

    private Package getCachedPackageConfig(Long packageId) {
        String cacheKey = PACKAGE_CONFIG_CACHE_PREFIX + packageId;
        return hybridCacheManager.get(cacheKey, () -> {
            return packageService.getPackageById(packageId);
        }, Duration.ofHours(1));
    }

    private String getCachedModelPermission(Long userId, Long modelId, Package packageConfig) {
        String cacheKey = MODEL_PERMISSION_CACHE_PREFIX + userId + ":" + modelId;
        return hybridCacheManager.get(cacheKey, () -> {
            return validateModelPermissionInternal(userId, modelId, packageConfig);
        }, Duration.ofMinutes(30));
    }

    private String getCachedDailyUsageValidation(Long userId, Long modelId) {
        String today = LocalDate.now().toString();
        String cacheKey = DAILY_USAGE_CACHE_PREFIX + userId + ":" + modelId + ":" + today;
        
        return hybridCacheManager.get(cacheKey, () -> {
            return validateDailyUsageInternal(userId, modelId);
        }, Duration.ofHours(1));
    }

}
```

#### 3.3.2 实现缓存更新策略
```java
@Component
@Slf4j
public class ChatLimitCacheManager {
    
    @Autowired
    private HybridCacheManager hybridCacheManager;
    
    /**
     * 用户套餐变更时清除相关缓存
     */
    @EventListener
    public void handlePackageChange(UserPackageChangeEvent event) {
        Long userId = event.getUserId();
        
        // 清除用户账户缓存
        hybridCacheManager.evict("user:account:" + userId);
        
        // 清除模型权限缓存
        hybridCacheManager.evictPattern("model:permission:" + userId + ":*");
        
        // 清除每日使用统计缓存
        hybridCacheManager.evictPattern("daily:usage:" + userId + ":*");
        
        log.info("用户{}套餐变更，已清除相关缓存", userId);
    }
    
    /**
     * 模型配置变更时清除相关缓存
     */
    @EventListener
    public void handleModelConfigChange(ModelConfigChangeEvent event) {
        Long modelId = event.getModelId();
        
        // 清除所有用户对该模型的权限缓存
        hybridCacheManager.evictPattern("model:permission:*:" + modelId);
        
        log.info("模型{}配置变更，已清除相关缓存", modelId);
    }
}
```

```java
@Component
@Slf4j
public class CacheMetricsCollector {
    
    private final MeterRegistry meterRegistry;
    private final Timer cacheGetTimer;
    private final Counter cacheHitCounter;
    private final Counter cacheMissCounter;
    
    public CacheMetricsCollector(MeterRegistry meterRegistry) {
        this.meterRegistry = meterRegistry;
        this.cacheGetTimer = Timer.builder("cache.get.time")
            .description("Cache get operation time")
            .register(meterRegistry);
        this.cacheHitCounter = Counter.builder("cache.hit")
            .description("Cache hit count")
            .register(meterRegistry);
        this.cacheMissCounter = Counter.builder("cache.miss")
            .description("Cache miss count")
            .register(meterRegistry);
    }
    
    public void recordCacheHit(String cacheName) {
        cacheHitCounter.increment(Tags.of("cache", cacheName));
    }
    
    public void recordCacheMiss(String cacheName) {
        cacheMissCounter.increment(Tags.of("cache", cacheName));
    }
    
    public <T> T timeGet(String cacheName, Supplier<T> supplier) {
        return cacheGetTimer.recordCallable(() -> supplier.get());
    }
}
```

```java
@Component
public class CacheHealthIndicator implements HealthIndicator {
    
    @Autowired
    private HybridCacheManager hybridCacheManager;
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    @Override
    public Health health() {
        try {
            // 检查Caffeine缓存
            boolean caffeineHealthy = checkCaffeineHealth();
            
            // 检查Redis缓存
            boolean redisHealthy = checkRedisHealth();
            
            if (caffeineHealthy && redisHealthy) {
                return Health.up()
                    .withDetail("caffeine", "UP")
                    .withDetail("redis", "UP")
                    .withDetail("timestamp", Instant.now())
                    .build();
            } else {
                return Health.down()
                    .withDetail("caffeine", caffeineHealthy ? "UP" : "DOWN")
                    .withDetail("redis", redisHealthy ? "UP" : "DOWN")
                    .withDetail("timestamp", Instant.now())
                    .build();
            }
        } catch (Exception e) {
            return Health.down()
                .withDetail("error", e.getMessage())
                .withDetail("timestamp", Instant.now())
                .build();
        }
    }
    
    private boolean checkCaffeineHealth() {
        try {
            // 简单的缓存操作测试
            String testKey = "health:check:caffeine";
            hybridCacheManager.putToCaffeine(testKey, "test", Duration.ofSeconds(10));
            Object result = hybridCacheManager.getFromCaffeine(testKey);
            return "test".equals(result);
        } catch (Exception e) {
            log.warn("Caffeine缓存健康检查失败", e);
            return false;
        }
    }
    
    private boolean checkRedisHealth() {
        try {
            // Redis连接测试
            String pong = redisTemplate.getConnectionFactory()
                .getConnection()
                .ping();
            return "PONG".equals(pong);
        } catch (Exception e) {
            log.warn("Redis缓存健康检查失败", e);
            return false;
        }
    }
}
```

```java
@RestController
@RequestMapping("/admin/cache")
@Slf4j
public class CacheManagementController {
    
    @Autowired
    private HybridCacheManager hybridCacheManager;
    
    /**
     * 获取缓存统计信息
     */
    @GetMapping("/stats")
    public Result<Map<String, Object>> getCacheStats() {
        Map<String, Object> stats = new HashMap<>();
        stats.put("caffeineStats", hybridCacheManager.getCaffeineStats());
        stats.put("redisStats", hybridCacheManager.getRedisStats());
        stats.put("timestamp", Instant.now());
        return Result.success(stats);
    }
    
    /**
     * 清除指定缓存
     */
    @DeleteMapping("/clear/{pattern}")
    public Result<String> clearCache(@PathVariable String pattern) {
        try {
            hybridCacheManager.evictPattern(pattern);
            log.info("清除缓存成功，模式: {}", pattern);
            return Result.success("缓存清除成功");
        } catch (Exception e) {
            log.error("清除缓存失败，模式: {}", pattern, e);
            return Result.failure("缓存清除失败: " + e.getMessage());
        }
    }
    
    /**
     * 预热指定缓存
     */
    @PostMapping("/warmup/{type}")
    public Result<String> warmupCache(@PathVariable String type) {
        try {
            switch (type) {
                case "user-account":
                    warmupUserAccountCache();
                    break;
                case "package-config":
                    warmupPackageConfigCache();
                    break;
                default:
                    return Result.failure("不支持的缓存类型: " + type);
            }
            return Result.success("缓存预热成功");
        } catch (Exception e) {
            log.error("缓存预热失败，类型: {}", type, e);
            return Result.failure("缓存预热失败: " + e.getMessage());
        }
```
