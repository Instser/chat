## 概述

本文档详细说明了系统中所有API接口的错误响应格式、错误码分类和前端处理建议。系统采用了两种不同的错误处理机制：

1. **聊天类API**: 错误作为聊天内容返回，保存到聊天记录
2. **非聊天类API**: 使用统一的错误响应格式，包含错误码和错误消息

## 错误响应格式

### 1. 聊天类API错误格式

**适用接口**:
- `/api/chat/stream` - 流式聊天
- `/api/chat/retry` - 重试聊天
- `/api/chat/edit-resend` - 编辑重发

**错误响应格式**:
```
data: {"role":"assistant","content":"Request failed. Please try again later."}
```

**特点**:
- 错误作为聊天消息返回
- 统一错误内容："Request failed. Please try again later."
- 自动保存到聊天记录数据库
- 前端在聊天界面中正常显示

### 2. 非聊天类API错误格式

**适用接口**: 除聊天类外的所有API接口

**错误响应格式**:
```json
{
  "code": 200,
  "message": "操作成功",
  "data": {
    "code": 6005,
    "message": "用户名或密码错误"
  },
  "timestamp": 1719648958000
}
```

**字段说明**:
- `code`: 固定为200（HTTP状态码）
- `message`: 固定为"操作成功"
- `data.code`: 具体的业务错误码
- `data.message`: 具体的错误描述
- `timestamp`: 响应时间戳

## 前端错误处理建议

### 1. 聊天类API错误处理

```javascript
// 流式聊天错误处理
const handleChatStream = (response) => {
  const reader = response.body.getReader();
  
  const readStream = async () => {
    const { done, value } = await reader.read();
    
    if (done) return;
    
    const chunk = new TextDecoder().decode(value);
    const lines = chunk.split('\n');
    
    for (const line of lines) {
      if (line.startsWith('data: ')) {
        try {
          const data = JSON.parse(line.slice(6));
          
          // 检查是否为错误消息
          if (data.role === 'assistant' && 
              data.content === 'Request failed. Please try again later.') {
            // 这是一个错误响应，但仍然作为聊天消息显示
            displayChatMessage(data);
            // 可以额外显示错误提示
            showErrorToast('聊天请求失败，请稍后重试');
          } else {
            // 正常的聊天消息
            displayChatMessage(data);
          }
        } catch (e) {
          console.error('解析聊天数据失败:', e);
        }
      }
    }
    
    readStream();
  };
  
  readStream();
};
```

### 2. 非聊天类API错误处理

```javascript
// 统一错误处理函数
const handleApiResponse = async (response) => {
  const data = await response.json();
  
  // 检查是否为错误响应
  if (data.data && data.data.code && data.data.code !== 200) {
    const errorCode = data.data.code;
    const errorMessage = data.data.message;
    
    // 根据错误码进行不同处理
    switch (Math.floor(errorCode / 1000)) {
      case 5: // 文件上传错误
        handleFileUploadError(errorCode, errorMessage);
        break;
      case 6: // 认证授权错误
        handleAuthError(errorCode, errorMessage);
        break;
      case 7: // 支付相关错误
        handlePaymentError(errorCode, errorMessage);
        break;
      case 8: // API接口错误
        handleApiError(errorCode, errorMessage);
        break;
      default:
        handleGenericError(errorCode, errorMessage);
    }
    
    return { success: false, error: { code: errorCode, message: errorMessage } };
  }
  
  // 成功响应
  return { success: true, data: data.data };
};

// 文件上传错误处理
const handleFileUploadError = (code, message) => {
  switch (code) {
    case 5002:
      showErrorToast('文件大小不能超过10MB');
      break;
    case 5003:
      showErrorToast('不支持该文件类型');
      break;
    case 5004:
      showErrorToast('文件解析失败，请检查文件格式');
      break;
    default:
      showErrorToast(message || '文件上传失败');
  }
};

// 认证授权错误处理
const handleAuthError = (code, message) => {
  switch (code) {
    case 6005:
      showErrorToast('用户名或密码错误');
      break;
    case 6006:
    case 6007:
      // Token过期或无效，跳转到登录页
      redirectToLogin();
      break;
    case 6003:
      showErrorToast('用户名已存在，请选择其他用户名');
      break;
    case 6004:
      showErrorToast('邮箱已存在，请使用其他邮箱');
      break;
    default:
      showErrorToast(message || '认证失败');
  }
};

// 支付相关错误处理
const handlePaymentError = (code, message) => {
  switch (code) {
    case 7002:
      showErrorToast('订单不存在，请重新创建订单');
      break;
    case 7007:
      showErrorToast('订单已过期，请重新创建订单');
      break;
    case 7008:
      showErrorToast('订单已支付，请勿重复支付');
      break;
    default:
      showErrorToast(message || '支付处理失败');
  }
};
```

```javascript
const ErrorToast = {
  // 显示错误提示
  show: (message, duration = 3000) => {

    toast.className = 'error-toast';
    toast.textContent = message;

    }, 100);

        document.body.removeChild(toast);
      }, 300);

  }

// 错误边界组件 (React示例)
class ErrorBoundary extends React.Component {
  constructor(props) {
    super(props);
    this.state = { hasError: false, error: null };
  }
  
  static getDerivedStateFromError(error) {
    return { hasError: true, error };
  }
  
  componentDidCatch(error, errorInfo) {
    console.error('错误边界捕获到错误:', error, errorInfo);
    
    // 可以上报错误到监控系统
    reportError(error, errorInfo);
  }
  
  render() {
    if (this.state.hasError) {
      return (
        <div className="error-fallback">
          <h2>出现了一些问题</h2>
          <p>请刷新页面重试，如果问题持续存在，请联系客服。</p>
          <button onClick={() => window.location.reload()}>
            刷新页面
          </button>
        </div>
      );
    }
    
    return this.props.children;
  }
}
```

```javascript
// 全局请求拦截器
const apiRequest = async (url, options = {}) => {
  const token = localStorage.getItem('authToken');
  
  const response = await fetch(url, {
    ...options,
    headers: {
      'Authorization': `Bearer ${token}`,
      'Content-Type': 'application/json',

    }
  });
  
  const result = await handleApiResponse(response);
  
  // 检查认证错误
  if (!result.success && [6006, 6007].includes(result.error.code)) {

    localStorage.removeItem('authToken');
    localStorage.removeItem('userInfo');
    window.location.href = '/login';
    return;
  }
  
  return result;
};
```

## 错误监控和上报

### 1. 错误统计

```javascript
// 错误统计
const ErrorTracker = {
  errors: new Map(),
  
  track: (errorCode, errorMessage, context = {}) => {
    const key = `${errorCode}-${errorMessage}`;
    const count = this.errors.get(key) || 0;
    
    this.errors.set(key, count + 1);
    
    // 上报到监控系统
    this.report(errorCode, errorMessage, context);
  },
  
  report: (errorCode, errorMessage, context) => {
    // 发送到错误监控服务
    fetch('/api/error-report', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        errorCode,
        errorMessage,
        context,
        timestamp: Date.now(),
        userAgent: navigator.userAgent,
        url: window.location.href
      })
    }).catch(console.error);
  }
};
```

### 2. 错误重试机制

```javascript
// 自动重试机制
const apiRequestWithRetry = async (url, options = {}, maxRetries = 3) => {
  let lastError;

  for (let i = 0; i <= maxRetries; i++) {
    try {
      const response = await fetch(url, options);
      const result = await handleApiResponse(response);

      if (result.success) {
        return result;
      }

      // 某些错误不需要重试
      const noRetryErrors = [6005, 6003, 6004, 7002, 7008];
      if (noRetryErrors.includes(result.error.code)) {
        return result;
      }

      lastError = result.error;

      // 等待后重试
      if (i < maxRetries) {
        await new Promise(resolve => setTimeout(resolve, 1000 * (i + 1)));
      }
    } catch (error) {
      lastError = { code: 500, message: error.message };

      if (i < maxRetries) {
        await new Promise(resolve => setTimeout(resolve, 1000 * (i + 1)));
      }
    }
  }

  return { success: false, error: lastError };
};
```

```javascript
// 全局错误状态管理 (Redux示例)
const errorReducer = (state = { errors: [] }, action) => {
  switch (action.type) {
    case 'ADD_ERROR':
      return {
        ...state,
        errors: [...state.errors, {
          id: Date.now(),
          code: action.payload.code,
          message: action.payload.message,
          timestamp: Date.now()
        }]
      };

    case 'REMOVE_ERROR':
      return {
        ...state,
        errors: state.errors.filter(error => error.id !== action.payload.id)
      };

    case 'CLEAR_ERRORS':
      return {
        ...state,
        errors: []
      };

    default:
      return state;
  }
};

// 错误处理Action
const addError = (code, message) => ({

  payload: { code, message }
});

const removeError = (id) => ({
  type: 'REMOVE_ERROR',
```
