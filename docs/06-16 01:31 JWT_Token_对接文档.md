## 📋 概述

本文档详细说明了前端如何与后端进行JWT Token认证对接，包括登录获取Token、请求携带Token、Token刷新等完整流程。

## 🔐 认证机制

### Token传递方式
系统支持两种Token传递方式：

1. **Cookie方式**（推荐）：登录后Token自动存储在Cookie中，浏览器自动携带
2. **Authorization Header方式**：手动在请求头中添加Token

### Token存储位置
- **Cookie名称**: `auth-token`
- **Cookie配置**: 
  - `HttpOnly=false` - 允许JavaScript访问
  - `Secure=false` - 开发环境支持HTTP
  - `SameSite=Lax` - 允许跨站请求携带
  - `Path=/api` - 仅在API路径下有效
  - `MaxAge=7天` - Token有效期

## 🚀 快速开始

### 1. 登录获取Token

```javascript
// 登录请求
const loginResponse = await fetch('http://localhost:8080/api/auth/login', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Origin': 'http://localhost:3000'
  },
  credentials: 'include', // 🔑 关键：允许Cookie传递
  body: JSON.stringify({
    usernameOrEmail: 'admin@example.com',
    password: 'admin123'
  })
});

const loginData = await loginResponse.json();

if (loginData.success) {
  // Token已自动存储在Cookie中
  console.log('登录成功:', loginData.data);
  
  // 可选：同时保存到localStorage作为备用
  localStorage.setItem('auth-token', loginData.data.token);
} else {
  console.error('登录失败:', loginData.message);
}
```

### 2. 发送认证请求

#### 方式1：Cookie自动传递（推荐）

```javascript
// 聊天请求示例
const chatResponse = await fetch('http://localhost:8080/api/chat', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Origin': 'http://localhost:3000'
  },
  credentials: 'include', // 🔑 关键：自动携带Cookie中的Token
  body: JSON.stringify({
    messages: [{ role: 'user', content: 'Hello' }],
    userId: 1,
    modelCode: 'qwen3-235b-a22b'
  })
});

if (chatResponse.ok) {
  // 处理响应...
} else if (chatResponse.status === 401) {
  // Token过期或无效，需要重新登录
  window.location.href = '/login';
}
```

#### 方式2：Authorization Header传递

```javascript
// 获取Token
const token = localStorage.getItem('auth-token');

const response = await fetch('http://localhost:8080/api/chat', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': `Bearer ${token}`, // 🔑 关键：在请求头中携带Token
    'Origin': 'http://localhost:3000'
  },
  credentials: 'include',
  body: JSON.stringify({
    messages: [{ role: 'user', content: 'Hello' }],
    userId: 1
  })
});
```

## 🛠️ API客户端封装

### 完整的API客户端类

```javascript
class ApiClient {
  constructor(baseURL = 'http://localhost:8080/api') {
    this.baseURL = baseURL;
  }

  /**
   * 通用请求方法
   */
  async request(endpoint, options = {}) {
    const url = `${this.baseURL}${endpoint}`;
    const token = localStorage.getItem('auth-token');
    
    const config = {
      headers: {
        'Content-Type': 'application/json',
        'Origin': 'http://localhost:3000',
        ...(token && { 'Authorization': `Bearer ${token}` }),
        ...options.headers
      },
      credentials: 'include', // 始终包含Cookie
      ...options
    };

    try {
      const response = await fetch(url, config);
      
      // 处理认证失败
      if (response.status === 401) {
        this.handleAuthError();
        throw new Error('认证失败，请重新登录');
      }
      
      return response;
    } catch (error) {
      console.error('API请求失败:', error);
      throw error;
    }
  }

  /**
   * 处理认证错误
   */
  handleAuthError() {
    localStorage.removeItem('auth-token');
    // 可以触发全局事件或重定向到登录页
    window.dispatchEvent(new CustomEvent('auth-error'));
  }

  /**
   * 登录
   */
  async login(credentials) {
    const response = await this.request('/auth/login', {
      method: 'POST',
      body: JSON.stringify(credentials)
    });
    
    const data = await response.json();
    if (data.success) {
      localStorage.setItem('auth-token', data.data.token);
    }
    return data;
  }

  /**
   * 登出
   */
  async logout() {
    try {
      await this.request('/auth/logout', { method: 'POST' });
    } finally {
      localStorage.removeItem('auth-token');
      // 清除Cookie（如果需要）
      document.cookie = 'auth-token=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/api;';
    }
  }

  /**
   * 聊天
   */
  async chat(messages, userId = 1, modelCode = 'qwen3-235b-a22b') {
    return this.request('/chat', {
      method: 'POST',
      body: JSON.stringify({ messages, userId, modelCode })
    });
  }

  /**
   * 获取用户信息
   */
  async getUserInfo() {
    const response = await this.request('/user/info');
    return response.json();
  }

  /**
   * 获取聊天历史
   */
  async getChatHistory(page = 1, size = 20) {
    const response = await this.request(`/chat/history?page=${page}&size=${size}`);
    return response.json();
  }
}

// 创建全局API客户端实例
const api = new ApiClient();

// 监听认证错误事件
window.addEventListener('auth-error', () => {
  alert('登录已过期，请重新登录');
  window.location.href = '/login';
});
```

## 🔄 使用示例

### React Hook示例

```javascript
import { useState, useEffect } from 'react';

// 自定义Hook：useAuth
export function useAuth() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    checkAuthStatus();
  }, []);

  const checkAuthStatus = async () => {
    try {
      const response = await api.getUserInfo();
      if (response.success) {
        setUser(response.data);
      }
    } catch (error) {
      console.log('未登录或Token已过期');
    } finally {
      setLoading(false);
    }
  };

  const login = async (credentials) => {
    const result = await api.login(credentials);
    if (result.success) {
      setUser(result.data);
    }
    return result;
  };

  const logout = async () => {
    await api.logout();
    setUser(null);
  };

  return { user, loading, login, logout };
}

// 使用示例
function App() {
  const { user, loading, login, logout } = useAuth();

  if (loading) return <div>Loading...</div>;

  return (
    <div>
      {user ? (
        <div>
          <h1>欢迎, {user.username}</h1>
          <button onClick={logout}>登出</button>
        </div>
      ) : (
        <LoginForm onLogin={login} />
      )}
    </div>
  );
}
```

## ⚠️ 重要注意事项

### 1. CORS配置
确保所有请求都包含正确的Origin头：
```javascript
headers: {
  'Origin': 'http://localhost:3000'  // 必须与后端CORS配置匹配
}
```

### 2. Credentials设置
所有需要认证的请求都必须设置：
```javascript
credentials: 'include'  // 允许Cookie传递
```

### 3. 错误处理
```javascript
// 统一错误处理
if (response.status === 401) {
  // Token过期或无效
  handleAuthError();
} else if (response.status === 403) {
  // 权限不足
  alert('权限不足');
} else if (!response.ok) {
  // 其他错误
  throw new Error(`HTTP ${response.status}: ${response.statusText}`);
}
```

### 4. Token刷新
当前系统Token有效期为7天，如需自动刷新功能，可以：
```javascript
// 检查Token是否即将过期
function isTokenExpiringSoon(token) {
  try {
    const payload = JSON.parse(atob(token.split('.')[1]));
    const expirationTime = payload.exp * 1000;
    const currentTime = Date.now();
    const timeUntilExpiration = expirationTime - currentTime;
    
    // 如果Token在1小时内过期，返回true
    return timeUntilExpiration < 60 * 60 * 1000;
  } catch (error) {
    return true; // 解析失败，认为需要刷新
  }
}
```
