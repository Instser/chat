```java
@Service
public class StreamingResponseService {

    private static final Logger log = LoggerFactory.getLogger(StreamingResponseService.class);
    private final SearchResultFormatter searchResultFormatter;
    private final CompletionFormatter completionFormatter;

    public StreamingResponseService(SearchResultFormatter searchResultFormatter,
                                  CompletionFormatter completionFormatter) {
        this.searchResultFormatter = searchResultFormatter;
        this.completionFormatter = completionFormatter;
    }

    /**
     * 发送联网搜索结果（简化版，无重试机制）
     */
    public boolean sendNetworkSearchResults(HttpServletResponse response, List<TavilySearchResult> results) {
        if (results == null || results.isEmpty()) {
            log.debug("无联网搜索结果需要发送");
            return true;
        }

        log.info("开始发送{}个联网搜索结果给前端", results.size());

        try {
            for (TavilySearchResult result : results) {
                String formattedResult = searchResultFormatter.formatAsStreamingData(result);
                writeToResponse(response, formattedResult);
            }

            log.info("成功发送{}个联网搜索结果给前端", results.size());
            return true;
        } catch (Exception e) {
            log.error("发送联网搜索结果失败", e);
            return false;
        }
    }

    /**
     * 发送完成标记
     */
    public boolean sendCompletion(HttpServletResponse response, CompletionData data) {
        try {
            String completionJson = completionFormatter.formatAsStreamingData(data);
            writeToResponse(response, completionJson);

            log.debug("发送完成标记成功");
            return true;
        } catch (Exception e) {
            log.error("发送完成标记失败", e);
            return false;
        }
    }

    private void writeToResponse(HttpServletResponse response, String data) throws IOException {
        response.getWriter().write(data);
        response.getWriter().flush();
    }
}
```

#### 1.2 线程本地搜索结果管理器 (NetworkSearchResultManager)
```java
@Component
public class NetworkSearchResultManager {

    private static final Logger log = LoggerFactory.getLogger(NetworkSearchResultManager.class);

    // 使用ThreadLocal存储当前线程的搜索结果
    private final ThreadLocal<List<TavilySearchResult>> currentSearchResults = new ThreadLocal<>();

    // 使用ThreadLocal存储发送状态，避免重复发送
    private final ThreadLocal<Boolean> resultsSent = new ThreadLocal<>();

    /**
     * 存储搜索结果到当前线程
     */
    public void storeSearchResults(List<TavilySearchResult> results) {
        if (results != null && !results.isEmpty()) {
            currentSearchResults.set(new ArrayList<>(results));
            resultsSent.set(false);
            log.debug("已存储{}个搜索结果到ThreadLocal", results.size());
        }
    }

    /**
     * 获取当前线程的搜索结果（只能获取一次）
     */
    public List<TavilySearchResult> getAndMarkSent() {
        List<TavilySearchResult> results = currentSearchResults.get();
        Boolean sent = resultsSent.get();

        if (results != null && !Boolean.TRUE.equals(sent)) {
            resultsSent.set(true);
            log.debug("获取{}个搜索结果并标记为已发送", results.size());
            return new ArrayList<>(results);
        }

        return Collections.emptyList();
    }

    /**
     * 检查是否有搜索结果
     */
    public boolean hasSearchResults() {
        List<TavilySearchResult> results = currentSearchResults.get();
        return results != null && !results.isEmpty();
    }

    /**
     * 清理当前线程的搜索结果
     */
    public void clearSearchResults() {
        currentSearchResults.remove();
        resultsSent.remove();
        log.debug("已清理ThreadLocal中的搜索结果");
    }
}
```

public class NetworkSearchData {

    private final List<TavilySearchResult> searchResults;
    private final long timestamp;

        this.sessionId = sessionId;
    }

    /**
     * 获取搜索结果的不可变副本

        return Collections.unmodifiableList(searchResults);
    }

     * 检查是否有搜索结果

    public boolean hasResults() {
        return !searchResults.isEmpty();
    }

    /**
     * 获取结果数量
     */
    public int getResultCount() {
        return searchResults.size();
    }

    /**
     * 检查数据是否过期（超过5分钟）
     */
    public boolean isExpired() {
        return System.currentTimeMillis() - timestamp > TimeUnit.MINUTES.toMillis(5);
    }

    // Getters
    public long getTimestamp() { return timestamp; }
    public String getSessionId() { return sessionId; }

    @Override
    public String toString() {
        return String.format("NetworkSearchData{sessionId='%s', resultCount=%d, timestamp=%d}",

@Component
public class SearchResultFormatter {
    
    private final ObjectMapper objectMapper = new ObjectMapper();
    
    /**
     * 格式化搜索结果为流式数据
     */
    public String formatAsStreamingData(TavilySearchResult result) {
        try {
            SearchResultData data = SearchResultData.from(result);
            String json = objectMapper.writeValueAsString(data);
            return String.format("u:%s\n", json);
        } catch (Exception e) {
            log.error("格式化搜索结果失败", e);
            throw new FormattingException("Failed to format search result", e);
        }
    }
    
    /**
     * 搜索结果数据传输对象
     */
    public static class SearchResultData {
        private String url;
        private String title;
        private String content;
        private String favicon;
        
        public static SearchResultData from(TavilySearchResult result) {
            SearchResultData data = new SearchResultData();
            data.url = result.getUrl() != null ? result.getUrl() : "";
            data.title = result.getTitle() != null ? result.getTitle() : "";
            
            // 限制content字段为前100个字符
            String content = result.getContent() != null ? result.getContent() : "";
            if (content.length() > 100) {
                content = content.substring(0, 100) + "...";
            }
            data.content = content;
            
            data.favicon = result.getFavicon() != null ? result.getFavicon() : "";
            return data;
        }
        
        // Getters and Setters
        public String getUrl() { return url; }
        public void setUrl(String url) { this.url = url; }
        public String getTitle() { return title; }
        public void setTitle(String title) { this.title = title; }
        public String getContent() { return content; }
        public void setContent(String content) { this.content = content; }
        public String getFavicon() { return favicon; }
        public void setFavicon(String favicon) { this.favicon = favicon; }
    }
}

@Component
public class CompletionFormatter {
    
    private final ObjectMapper objectMapper = new ObjectMapper();

     * 格式化完成数据为流式数据
     */
    public String formatAsStreamingData(CompletionData data) {
        try {
            Map<String, Object> completionMap = new HashMap<>();
            completionMap.put("finishReason", data.getFinishReason());
            completionMap.put("usage", Map.of(
                "promptTokens", data.getPromptTokens(),
                "completionTokens", data.getCompletionTokens()
            ));
            
            if (data.getTitle() != null && !data.getTitle().trim().isEmpty()) {
                completionMap.put("title", data.getTitle().trim());
            } else {
                completionMap.put("title", false);
            }
            
            String json = objectMapper.writeValueAsString(completionMap);
            return String.format("d:%s\n", json);
        } catch (Exception e) {
            log.error("格式化完成数据失败", e);
            throw new FormattingException("Failed to format completion data", e);
        }
    }
}

## 🚨 **ThreadLocal线程问题修正**

### 发现的严重问题

#### **线程模型不匹配问题**
经过深入分析发现：
- `processNetworkContent` 在**主HTTP请求线程**中执行
- `handleStreamCompletionWithInterruptStatus` 在**流式响应处理线程**中执行
- **ThreadLocal数据无法跨线程传递！**

#### **当前实现的线程问题**
```java
// 主线程：存储数据到ThreadLocal
processNetworkContent() -> ThreadLocal.set(searchResults)

// 异步线程：尝试获取数据（获取不到！）
handleStreamCompletionWithInterruptStatus() -> ThreadLocal.get() // 返回null
```

### 修正方案：会话级状态管理

#### 1. 使用会话ID作为键的状态管理
**原问题**: ThreadLocal无法跨线程传递数据
**修正方案**: 使用ConcurrentHashMap + 会话ID进行状态管理

### 改进方案的核心实现

#### 修正1: 会话级状态管理（解决跨线程问题）
```java
@Component
public class NetworkSearchResultManager {

    private static final Logger log = LoggerFactory.getLogger(NetworkSearchResultManager.class);

    // 使用会话ID作为键的状态管理，解决跨线程问题
    private final ConcurrentHashMap<String, NetworkSearchData> sessionSearchData = new ConcurrentHashMap<>();

    // 定时清理过期数据
    private final ScheduledExecutorService cleanupExecutor = Executors.newSingleThreadScheduledExecutor();

    @PostConstruct
    public void init() {
        // 每5分钟清理一次过期数据
        cleanupExecutor.scheduleAtFixedRate(this::cleanupExpiredData, 5, 5, TimeUnit.MINUTES);
    }

    /**
     * 存储搜索结果（跨线程安全）
     */
    public void storeSearchResults(List<TavilySearchResult> results, String sessionId) {
        if (results != null && !results.isEmpty() && sessionId != null) {
            NetworkSearchData data = new NetworkSearchData(results, sessionId);
            sessionSearchData.put(sessionId, data);
            log.debug("已存储{}个搜索结果到会话：{}", results.size(), sessionId);
        }
    }

    /**
     * 获取并清理搜索结果（跨线程安全）
     */
    public List<TavilySearchResult> getAndClear(String sessionId) {
        if (sessionId == null) {
            return Collections.emptyList();
        }

        NetworkSearchData data = sessionSearchData.remove(sessionId);
        if (data != null && data.hasResults() && !data.isExpired()) {
            log.debug("获取{}个搜索结果从会话：{}", data.getResultCount(), sessionId);
            return data.getSearchResults();
        }

        return Collections.emptyList();
    }

    /**
     * 强制清理会话数据
     */
    public void forceCleanup(String sessionId) {
        if (sessionId != null) {
            sessionSearchData.remove(sessionId);
            log.debug("强制清理会话数据：{}", sessionId);
        }
    }

    /**
     * 清理过期数据
     */
    private void cleanupExpiredData() {
        long now = System.currentTimeMillis();
        sessionSearchData.entrySet().removeIf(entry -> {
            NetworkSearchData data = entry.getValue();
            if (data.isExpired()) {
                log.debug("清理过期搜索数据，会话：{}", entry.getKey());
                return true;
            }
            return false;
        });
    }

    @PreDestroy
    public void destroy() {
        cleanupExecutor.shutdown();
        sessionSearchData.clear();
    }
}
```

    private final ConcurrentHashMap<String, NetworkSearchData> sessionSearchData = new ConcurrentHashMap<>();

    /**
     * 获取并清理（原子操作，确保只能调用一次）
     */
    public List<TavilySearchResult> getAndClear(String sessionId) {
        if (sessionId == null) {
            return Collections.emptyList();
        }

        NetworkSearchData data = sessionSearchData.remove(sessionId); // 原子操作
        if (data != null && data.hasResults() && !data.isExpired()) {
            return data.getSearchResults();
        }
        return Collections.emptyList();
    }
}
