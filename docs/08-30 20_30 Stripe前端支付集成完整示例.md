## 📋 后端API接口文档

### 认证相关API

#### 用户登录
```http
POST /api/auth/login
Content-Type: application/json

{
  "usernameOrEmail": "admin",
  "password": "admin123"
}
```

**响应示例:**
```json
{
  "code": 200,
  "message": "登录成功",
  "data": {
    "id": "1",
    "username": "admin",
    "email": "admin@example.com",
    "role": "admin",
    "status": 1
  }
}
```

### 支付相关API

#### 获取套餐列表
```http
GET /api/payment/packages
Cookie: auth-token=<JWT_TOKEN>
```

**响应示例:**
```json
{
  "code": 200,
  "message": "操作成功",
  "data": [
    {
      "id": 3,
      "name": "Pro",
      "description": "Research, code, and organize",
      "price": 5.00,
      "currency": "USD",
      "period": 30,
      "features": [
        "All features of the free package",
        "Almost unlimited requests for performance model calls",
        "20 best model call requests per day"
      ]
    }
  ]
}
```

#### 创建支付订单
```http
POST /api/payment/create
Content-Type: application/json
Cookie: auth-token=<JWT_TOKEN>

{
  "packageId": 3,
  "paymentMethod": "stripe",
  "currency": "USD"
}
```

**响应示例:**
```json
{
  "code": 200,
  "message": "支付订单创建成功",
  "data": {
    "orderNo": "JY202508302035465565",
    "amount": 10.00,
    "currency": "USD",
    "success": true,
    "paymentMethod": "stripe",
    "environment": "测试环境",
    "client_secret": "pi_3S1o1oRWJJRQKPQG1xjHlMXa_secret_Yb41mTYl5qmpO40C6YZ77RoUg",
    "publishable_key": "pk_test_your_stripe_key",
    "payment_intent_id": "pi_3S1o1oRWJJRQKPQG1xjHlMXa",
    "payment_method_types": ["card", "google_pay", "apple_pay"],
    "requires_action": false
  }
}
```

#### 查询订单状态
```http
GET /api/payment/status/{orderNo}
Cookie: auth-token=<JWT_TOKEN>
```

**响应示例:**
```json
{
  "code": 200,
  "message": "操作成功",
  "data": {
    "status": 1,
    "message": "支付成功",
    "orderNo": "JY202508302035465565",
    "amount": 10.00,
    "currency": "USD",
    "paymentMethod": "stripe"
  }
}
```

#### 同步订单状态
```http
POST /api/payment/sync/{orderNo}
Cookie: auth-token=<JWT_TOKEN>
```

    <script>
        // 配置信息
        const CONFIG = {
            API_BASE: 'http://localhost:8080/api',
            STRIPE_PUBLIC_KEY: 'pk_test_your_stripe_key'
        };

        // 全局变量
        let stripe = null;
        let selectedPackage = null;
        let currentOrder = null;

        // 初始化
        document.addEventListener('DOMContentLoaded', async () => {
            // 初始化Stripe
            stripe = Stripe(CONFIG.STRIPE_PUBLIC_KEY);
            
            // 检查登录状态并加载数据
            await checkLoginStatus();
        });

        // 检查登录状态
        async function checkLoginStatus() {
            try {
                const response = await fetch(`${CONFIG.API_BASE}/payment/packages`, {
                    method: 'GET',
                    credentials: 'include'
                });

                if (response.ok) {
                    const result = await response.json();
                    if (result.code === 200) {
                        document.getElementById('loginStatus').innerHTML = 
                            '<span style="color: #28a745;">✅ 已登录，正在加载套餐...</span>';
                        await loadPackages();
                        return;
                    }
                }
                
                // 未登录
                document.getElementById('loginStatus').innerHTML = 
                    '<span style="color: #dc3545;">❌ 未登录，请先登录</span>';
                document.getElementById('loginBtn').classList.remove('hidden');
                
            } catch (error) {
                console.error('检查登录状态失败:', error);
                document.getElementById('loginStatus').innerHTML = 
                    '<span style="color: #dc3545;">❌ 检查登录状态失败</span>';
                document.getElementById('loginBtn').classList.remove('hidden');
            }
        }

        // 处理登录
        async function handleLogin() {
            try {
                const response = await fetch(`${CONFIG.API_BASE}/auth/login`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    credentials: 'include',
                    body: JSON.stringify({
                        usernameOrEmail: 'admin',
                        password: 'admin123'
                    })
                });

                const result = await response.json();
                if (result.code === 200) {
                    document.getElementById('loginStatus').innerHTML = 
                        '<span style="color: #28a745;">✅ 登录成功，正在加载套餐...</span>';
                    document.getElementById('loginBtn').classList.add('hidden');
                    await loadPackages();
                } else {
                    document.getElementById('loginStatus').innerHTML = 
                        `<span style="color: #dc3545;">❌ 登录失败: ${result.message}</span>`;
                }
            } catch (error) {
                console.error('登录失败:', error);
                document.getElementById('loginStatus').innerHTML = 
                    '<span style="color: #dc3545;">❌ 登录请求失败</span>';
            }
        }

        // 加载套餐列表
        async function loadPackages() {
            try {
                const response = await fetch(`${CONFIG.API_BASE}/payment/packages`, {
                    method: 'GET',
                    credentials: 'include'
                });

                const result = await response.json();
                if (result.code === 200) {
                    displayPackages(result.data);
                } else {
                    document.getElementById('packagesContainer').innerHTML = 
                        `<div class="status error">加载套餐失败: ${result.message}</div>`;
                }
            } catch (error) {
                console.error('加载套餐失败:', error);
                document.getElementById('packagesContainer').innerHTML = 
                    '<div class="status error">加载套餐失败，请刷新重试</div>';
            }
        }

        // 显示套餐列表
        function displayPackages(packages) {
            const container = document.getElementById('packagesContainer');
            container.innerHTML = '';

            packages.forEach(pkg => {
                const packageCard = document.createElement('div');
                packageCard.className = 'package-card';
                packageCard.onclick = () => selectPackage(pkg);
                
                const periodText = pkg.period === -1 ? '永久' : `${pkg.period}天`;
                
                packageCard.innerHTML = `
                    <div class="package-name">${pkg.name}</div>
                    <div class="package-price">$${pkg.price.toFixed(2)}</div>
                    <div class="package-period">${periodText}</div>
                    <div style="margin-top: 10px; font-size: 12px; color: #6c757d;">
                        ${pkg.description}
                    </div>
                `;
                
                container.appendChild(packageCard);
            });
        }

        // 选择套餐
        function selectPackage(pkg) {
            // 移除之前的选中状态
            document.querySelectorAll('.package-card').forEach(card => {
                card.classList.remove('selected');
            });
            
            // 添加选中状态
            event.currentTarget.classList.add('selected');
            selectedPackage = pkg;
            
            // 显示支付表单
            document.getElementById('paymentSection').classList.remove('hidden');
            document.getElementById('payBtn').disabled = false;
            
            console.log('选择套餐:', pkg);
        }

        // 处理支付
        async function handlePayment() {
            if (!selectedPackage) {
                alert('请先选择套餐');
                return;
            }

            const payBtn = document.getElementById('payBtn');
            const originalText = payBtn.innerHTML;
            
            try {
                payBtn.innerHTML = '<span class="loading"></span>创建订单中...';
                payBtn.disabled = true;

                const response = await fetch(`${CONFIG.API_BASE}/payment/create`, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    credentials: 'include',
                    body: JSON.stringify({
                        packageId: selectedPackage.id,
                        paymentMethod: 'stripe',
                        currency: 'USD'
                    })
                });

                const result = await response.json();
                
                if (result.code === 200) {
                    currentOrder = result.data;
                    
                    // 显示订单信息
                    const statusSection = document.getElementById('statusSection');
                    statusSection.classList.remove('hidden');
                    
                    document.getElementById('paymentStatus').innerHTML = `
                        <div class="status success">
                            <strong>✅ 支付订单创建成功！</strong><br>
                            订单号: ${currentOrder.orderNo}<br>
                            金额: $${currentOrder.amount} ${currentOrder.currency}<br>
                            PaymentIntent ID: ${currentOrder.payment_intent_id}
                        </div>
                    `;
                    
                    // 显示确认支付按钮
                    document.getElementById('confirmPayBtn').classList.remove('hidden');
                    payBtn.innerHTML = '订单已创建';
                    
                } else {
                    document.getElementById('paymentStatus').innerHTML = `
                        <div class="status error">
                            ❌ 创建订单失败: ${result.message}
                        </div>
                    `;
                    payBtn.innerHTML = originalText;
                    payBtn.disabled = false;
                }
                
            } catch (error) {
                console.error('创建订单失败:', error);
                document.getElementById('paymentStatus').innerHTML = `
                    <div class="status error">
                        ❌ 创建订单失败: ${error.message}
                    </div>
                `;
                payBtn.innerHTML = originalText;
                payBtn.disabled = false;
            }
        }

        // 确认支付
        async function confirmPayment() {
            if (!currentOrder || !currentOrder.client_secret) {
                alert('无效的订单信息');
                return;
            }

            const confirmBtn = document.getElementById('confirmPayBtn');
            const originalText = confirmBtn.innerHTML;
            
            try {
                confirmBtn.innerHTML = '<span class="loading"></span>支付处理中...';
                confirmBtn.disabled = true;

                // 获取选择的测试卡号
                const cardNumber = document.getElementById('testCardSelect').value;
                const cardholderName = document.getElementById('cardholderName').value;
                const cardholderEmail = document.getElementById('cardholderEmail').value;

                const { error, paymentIntent } = await stripe.confirmCardPayment(
                    currentOrder.client_secret,
                    {
                        payment_method: {
                            card: {
                                number: cardNumber,
                                exp_month: 12,
                                exp_year: 2030,
                                cvc: '123'
                            },
                            billing_details: {
                                name: cardholderName,
                                email: cardholderEmail
                            }
                        }
                    }
                );

                if (error) {
                    document.getElementById('paymentStatus').innerHTML = `
                        <div class="status error">
                            <strong>❌ 支付失败</strong><br>
                            错误信息: ${error.message}<br>
                            错误类型: ${error.type}<br>
                            ${error.code ? `错误代码: ${error.code}` : ''}
                        </div>
                    `;
                    confirmBtn.innerHTML = originalText;
                    confirmBtn.disabled = false;
                } else {
                    document.getElementById('paymentStatus').innerHTML = `
                        <div class="status success">
                            <strong>🎉 支付成功！</strong><br>
                            PaymentIntent ID: ${paymentIntent.id}<br>
                            状态: ${paymentIntent.status}<br>
                            金额: $${paymentIntent.amount / 100} ${paymentIntent.currency.toUpperCase()}<br>
                            支付方法: ${paymentIntent.payment_method}
                        </div>
                    `;
                    confirmBtn.innerHTML = '✅ 支付完成';
                    document.getElementById('checkStatusBtn').classList.remove('hidden');
                    
                    // 3秒后自动查询订单状态
                    setTimeout(() => {
                        checkOrderStatus();
                    }, 3000);
                }

            } catch (error) {
                console.error('支付处理失败:', error);
                document.getElementById('paymentStatus').innerHTML = `
                    <div class="status error">
                        ❌ 支付处理失败: ${error.message}
                    </div>
                `;
                confirmBtn.innerHTML = originalText;
                confirmBtn.disabled = false;
            }
        }

        // 查询订单状态
        async function checkOrderStatus() {
            if (!currentOrder || !currentOrder.orderNo) {
                alert('无效的订单号');
                return;
            }

            try {
                const response = await fetch(`${CONFIG.API_BASE}/payment/status/${currentOrder.orderNo}`, {
                    method: 'GET',
                    credentials: 'include'
                });

                const result = await response.json();
                
                if (result.code === 200) {
                    const statusHtml = document.getElementById('paymentStatus').innerHTML;
                    document.getElementById('paymentStatus').innerHTML = statusHtml + `
                        <div class="status info" style="margin-top: 15px;">
                            <strong>📋 订单状态查询结果</strong><br>
                            状态: ${result.data.message}<br>
                            订单号: ${result.data.orderNo}
                        </div>
                    `;
                } else {
                    console.error('查询订单状态失败:', result.message);
                }
                
            } catch (error) {
                console.error('查询订单状态失败:', error);
            }
        }
    </script>
</body>
</html>

## 🎯 核心功能特点

### 🔐 完整的认证流程
- **自动检测登录状态** - 页面加载时自动检查用户登录状态
- **一键登录功能** - 内置admin账户快速登录
- **Cookie自动管理** - 自动处理JWT Token的存储和传递

### 📦 动态套餐管理
- **实时套餐加载** - 从后端API动态获取套餐列表
- **可视化套餐选择** - 卡片式界面，支持鼠标悬停和选中效果
- **价格信息展示** - 清晰显示套餐价格、周期和描述

### 💳 完整Stripe集成
- **自动加载Stripe.js** - CDN方式加载，无需本地依赖
- **多种测试场景** - 支持成功、失败、余额不足等多种测试卡号
- **实时支付反馈** - 支付过程中的状态更新和错误处理
- **支付确认流程** - 两步式支付：创建订单 → 确认支付

### 🔍 订单状态管理
- **订单创建确认** - 显示订单号、金额、PaymentIntent ID
- **支付状态查询** - 支持手动和自动状态查询
- **完整错误处理** - 详细的错误信息显示和处理

## 🚀 使用方法

### 部署步骤
1. **保存代码** - 将HTML代码保存为 `stripe-payment.html` 文件
2. **启动后端** - 确保后端服务运行在 `http://localhost:8080`
3. **打开页面** - 在浏览器中打开HTML文件
4. **开始测试** - 按照界面流程进行支付测试

### 测试流程
1. **登录认证** - 自动检测或手动登录
2. **选择套餐** - 点击套餐卡片进行选择
3. **填写信息** - 输入持卡人信息和选择测试卡号
4. **创建订单** - 点击创建支付订单按钮
5. **确认支付** - 使用Stripe进行支付确认
6. **查看结果** - 查看支付结果和订单状态

## 🔍 故障排除

### 常见问题

**1. 跨域问题**
- 确保后端CORS配置正确
- 检查 `credentials: 'include'` 设置

**2. 支付失败**
- 确认Stripe密钥配置正确
- 检查网络连接和控制台错误

**3. 认证失败**
- 确认Cookie设置正确
- 检查JWT Token有效性

### 调试建议
- **开启浏览器开发者工具** - 查看网络请求和控制台错误
- **检查后端日志** - 查看服务器端的错误信息
- **验证API响应** - 确认后端API返回格式正确
