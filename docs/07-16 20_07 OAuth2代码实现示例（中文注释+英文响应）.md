### OAuth2Service接口
```java
/**
 * OAuth2登录服务接口
 * 提供Google和GitHub第三方登录功能
 * 
 * @author instser
 * @since 2025-07-16
 */
public interface OAuth2Service {
    
    /**
     * 处理OAuth2登录流程
     * 
     * @param userInfo OAuth2用户信息
     * @return 登录用户对象
     * @throws OAuth2AuthenticationException 认证失败时抛出
     */
    User processOAuth2Login(OAuth2UserInfo userInfo);
    
    /**
     * 绑定OAuth2账号到现有用户
     * 
     * @param userId 用户ID
     * @param userInfo OAuth2用户信息
     */
    void bindOAuth2Account(Long userId, OAuth2UserInfo userInfo);
    
    /**
     * 解绑OAuth2账号
     * 
     * @param userId 用户ID
     * @param provider OAuth2提供商（google/github）
     */
    void unbindOAuth2Account(Long userId, String provider);
}
```

### OAuth2ServiceImpl实现类
```java
/**
 * OAuth2登录服务实现类
 * 
 * @author instser
 * @since 2025-07-16
 */
@Service
@Slf4j
@RequiredArgsConstructor
public class OAuth2ServiceImpl implements OAuth2Service {
    
    private final UserService userService;
    private final UserOAuthMapper userOAuthMapper;
    private final TokenEncryptionService tokenEncryptionService;
    
    @Override
    @Transactional
    public User processOAuth2Login(OAuth2UserInfo userInfo) {
        log.info("【OAuth2登录】开始处理登录请求，提供商：{}，用户ID：{}", 
                userInfo.getProvider(), userInfo.getId());
        
        try {
            // 1. 查找已绑定的OAuth账号
            UserOAuth existingOAuth = findByProviderAndProviderId(
                userInfo.getProvider(), userInfo.getId());
            
            if (existingOAuth != null) {
                // 已绑定账号，直接登录
                User user = userService.getById(existingOAuth.getUserId());
                if (user != null && user.getStatus() == 1) {
                    updateLastLoginTime(existingOAuth.getId());
                    log.info("【OAuth2登录成功】已绑定用户登录，用户ID：{}", user.getId());
                    return user;
                } else {
                    log.warn("【OAuth2登录失败】绑定的用户账号异常，用户ID：{}", existingOAuth.getUserId());
                    throw new OAuth2AuthenticationException("Account is disabled or not found");
                }
            }
            
            // 2. 检查邮箱是否已存在
            User existingUser = userService.getUserByEmail(userInfo.getEmail());
            if (existingUser != null) {
                // 邮箱已存在，需要账号合并确认
                log.info("【OAuth2登录】检测到邮箱冲突，需要账号合并，邮箱：{}", userInfo.getEmail());
                throw new AccountMergeRequiredException("An account with this email already exists. Please confirm account merge.", existingUser, userInfo);
            }
            
            // 3. 创建新用户
            User newUser = createUserFromOAuth(userInfo);
            bindOAuth2Account(newUser.getId(), userInfo);
            
            log.info("【OAuth2登录成功】创建新用户并绑定OAuth账号，用户ID：{}", newUser.getId());
            return newUser;
            
        } catch (AccountMergeRequiredException e) {
            // 重新抛出账号合并异常，由控制器处理
            throw e;
        } catch (Exception e) {
            log.error("【OAuth2登录失败】处理登录请求时发生异常，提供商：{}，错误：{}", 
                    userInfo.getProvider(), e.getMessage(), e);
            throw new OAuth2AuthenticationException("Login failed. Please try again later.");
        }
    }
    
    /**
     * 从OAuth信息创建新用户
     * 
     * @param userInfo OAuth2用户信息
     * @return 创建的用户对象
     */
    private User createUserFromOAuth(OAuth2UserInfo userInfo) {
        // 生成唯一用户名
        String username = generateUniqueUsername(userInfo.getUsername(), userInfo.getProvider());
        // 生成随机密码（OAuth用户不使用密码登录）
        String randomPassword = UUID.randomUUID().toString();
        
        log.debug("【创建OAuth用户】用户名：{}，邮箱：{}，提供商：{}", 
                username, userInfo.getEmail(), userInfo.getProvider());
        
        return userService.registerUser(username, userInfo.getEmail(), randomPassword);
    }
    
    /**
     * 生成唯一用户名
     * 
     * @param baseUsername 基础用户名
     * @param provider OAuth提供商
     * @return 唯一用户名
     */
    private String generateUniqueUsername(String baseUsername, String provider) {
        String username = baseUsername + "_" + provider;
        int counter = 1;
        
        // 确保用户名唯一
        while (userService.getUserByUsername(username) != null) {
            username = baseUsername + "_" + provider + "_" + counter;
            counter++;
        }
        
        log.debug("【生成用户名】原始用户名：{}，最终用户名：{}", baseUsername, username);
        return username;
    }
}
```

### OAuth2Controller控制器
```java
/**
 * OAuth2认证控制器
 * 处理Google和GitHub第三方登录
 * 
 * @author instser
 * @since 2025-07-16
 */
@RestController
@RequestMapping("/auth/oauth2")
@Slf4j
@RequiredArgsConstructor
public class OAuth2Controller {
    
    private final OAuth2Service oauth2Service;
    private final JwtUtil jwtUtil;
    private final OAuth2ErrorHandler errorHandler;
    
    /**
     * 发起OAuth2授权
     * 
     * @param provider OAuth提供商（google/github）
     * @param request HTTP请求对象
     * @return 授权URL响应
     */
    @GetMapping("/authorize/{provider}")
    public ResponseEntity<?> authorize(@PathVariable String provider, HttpServletRequest request) {
        log.info("【OAuth2授权】用户发起OAuth2授权，提供商：{}", provider);
        
        try {
            // 构建授权URL
            String authorizationUrl = buildAuthorizationUrl(provider, request);
            
            Map<String, String> response = new HashMap<>();
            response.put("authorizationUrl", authorizationUrl);
            response.put("provider", provider);
            
            log.debug("【OAuth2授权】授权URL生成成功，提供商：{}", provider);
            return ResponseEntity.ok(Result.success(response, "Authorization URL generated successfully"));
            
        } catch (Exception e) {
            log.error("【OAuth2授权失败】生成授权URL失败，提供商：{}，错误：{}", provider, e.getMessage(), e);
            return ResponseEntity.ok(Result.error("Failed to generate authorization URL"));
        }
    }
    
    /**
     * 处理OAuth2回调
     * 
     * @param provider OAuth提供商
     * @param code 授权码
     * @param state 状态参数
     * @param response HTTP响应对象
     * @return 重定向响应
     */
    @GetMapping("/callback/{provider}")
    public ResponseEntity<?> callback(@PathVariable String provider,
                                    @RequestParam String code,
                                    @RequestParam(required = false) String state,
                                    HttpServletResponse response) {
        log.info("【OAuth2回调】收到OAuth2回调请求，提供商：{}，授权码：{}", provider, code.substring(0, 10) + "...");
        
        try {
            // 1. 使用授权码获取访问令牌
            OAuth2AccessToken accessToken = exchangeCodeForToken(provider, code);
            log.debug("【OAuth2回调】访问令牌获取成功，提供商：{}", provider);
            
            // 2. 使用访问令牌获取用户信息
            OAuth2UserInfo userInfo = getUserInfo(provider, accessToken.getTokenValue());
            log.debug("【OAuth2回调】用户信息获取成功，提供商：{}，用户邮箱：{}", provider, userInfo.getEmail());
            
            // 3. 处理登录逻辑
            User user = oauth2Service.processOAuth2Login(userInfo);
            
            // 4. 生成JWT令牌
            String jwtToken = jwtUtil.generateToken(user);
            log.debug("【OAuth2回调】JWT令牌生成成功，用户ID：{}", user.getId());
            
            // 5. 设置安全Cookie
            response.addHeader(HttpHeaders.SET_COOKIE, jwtUtil.createAuthCookie(jwtToken).toString());
            
            // 6. 重定向到前端成功页面
            String redirectUrl = "http://localhost:3000/dashboard?token=" + jwtToken;
            log.info("【OAuth2登录成功】用户登录成功，用户ID：{}，重定向到：{}", user.getId(), redirectUrl);
            
            return ResponseEntity.status(HttpStatus.FOUND)
                    .header(HttpHeaders.LOCATION, redirectUrl)
                    .build();
                    
        } catch (AccountMergeRequiredException e) {
            // 账号合并异常 - 需要用户确认
            log.warn("【OAuth2回调】检测到账号冲突，需要合并确认，邮箱：{}", e.getOAuthInfo().getEmail());
            String errorUrl = "http://localhost:3000/merge-confirm?email=" + 
                URLEncoder.encode(e.getOAuthInfo().getEmail(), StandardCharsets.UTF_8) +
                "&provider=" + provider;
            return ResponseEntity.status(HttpStatus.FOUND)
                    .header(HttpHeaders.LOCATION, errorUrl)
                    .build();
                    
        } catch (OAuth2AuthenticationException e) {
            // OAuth认证异常
            log.error("【OAuth2回调失败】认证失败，提供商：{}，错误：{}", provider, e.getMessage());
            String errorUrl = "http://localhost:3000/login?error=" + 
                URLEncoder.encode("Authentication failed. Please try again.", StandardCharsets.UTF_8);
            return ResponseEntity.status(HttpStatus.FOUND)
                    .header(HttpHeaders.LOCATION, errorUrl)
                    .build();
                    
        } catch (Exception e) {
            // 其他系统异常
            log.error("【OAuth2回调失败】系统异常，提供商：{}，错误：{}", provider, e.getMessage(), e);
            String errorUrl = "http://localhost:3000/login?error=" + 
                URLEncoder.encode("Service temporarily unavailable. Please try again later.", StandardCharsets.UTF_8);
            return ResponseEntity.status(HttpStatus.FOUND)
                    .header(HttpHeaders.LOCATION, errorUrl)
                    .build();
        }
    }
}
```

### 错误处理器
```java
/**
 * OAuth2错误处理器
 * 提供用户友好的英文错误消息
 * 
 * @author instser
 * @since 2025-07-16
 */
@Component
public class OAuth2ErrorHandler {
    
    /**
     * 获取用户友好的错误消息（英文）
     * 
     * @param errorCode 错误码
     * @param provider OAuth提供商
     * @return 英文错误消息
     */
    public String getErrorMessage(String errorCode, String provider) {
        String providerName = getProviderDisplayName(provider);
        
        Map<String, String> errorMessages = Map.of(
            "access_denied", "Authorization was cancelled. Please try again if you want to login with " + providerName,
            "invalid_request", "Invalid login request. Please refresh the page and try again",
            "server_error", providerName + " service is temporarily unavailable. Please try again later or use email login",
            "temporarily_unavailable", providerName + " service is busy. Please try again later",
            "network_error", "Network connection error. Please check your connection and try again",
            "rate_limit_exceeded", "Too many login attempts. Please wait a moment and try again",
            "token_expired", "Login session expired. Please start the login process again"
        );
        
        return errorMessages.getOrDefault(errorCode, 
            "Login failed. Please try again or contact support if the problem persists");
    }
    
    /**
     * 获取提供商显示名称
     * 
     * @param provider 提供商代码
     * @return 显示名称
     */
    private String getProviderDisplayName(String provider) {
        return switch (provider.toLowerCase()) {
            case "google" -> "Google";
            case "github" -> "GitHub";
            default -> provider;
        };
    }
    
    /**
     * 创建标准错误响应
     * 
     * @param errorCode 错误码
     * @param provider OAuth提供商
     * @return 错误响应对象
     */
    public Result<Object> createErrorResponse(String errorCode, String provider) {
        String message = getErrorMessage(errorCode, provider);
        return Result.error(message);
    }
}
```

## 📝 关键要点总结

### ✅ 正确做法
1. **代码注释**: 使用中文，如 `// 1. 查找已绑定的OAuth账号`
2. **日志记录**: 使用中文，如 `log.info("【OAuth2登录成功】创建新用户")`
3. **方法文档**: 使用中文JavaDoc注释
4. **API响应**: 使用英文消息，如 `"Login failed. Please try again later."`

### ❌ 避免做法
1. 不要在代码注释中使用英文
2. 不要在日志记录中使用英文
3. 不要在API响应消息中使用中文
4. 不要混合使用中英文

### 🎯 实施建议
1. **团队协作**: 中文注释便于团队理解和维护
2. **国际化**: 英文响应消息符合国际化标准
3. **用户体验**: 英文错误提示更专业
4. **代码质量**: 保持注释和响应消息的一致性

---
**重要提醒**: 这种实现方式既保持了代码的可读性和可维护性（中文注释），又确保了API的国际化标准（英文响应），是最佳的实践方案。
