# 前端支付相关API文档

## 概述

本文档描述了前端项目中所有与支付(payment)和套餐(package)相关的API接口调用方法，基于TypeScript实现。

## 基础信息

- **API基础路径**: `/api/payment`
- **前端API库**: `@/lib/api`
- **认证方式**: 自动处理JWT Token
- **数据格式**: JSON

## 数据类型定义

### Package (套餐信息)

```typescript
interface Package {
  id: number;           // 套餐ID
  name: string;         // 套餐名称
  description: string;  // 套餐描述
  price: number;        // 价格
  currency: string;     // 货币类型 (USD/CNY)
  period: number;       // 有效期天数，如30表示30天
  features: string[];   // 功能特性列表
}
```

### PaymentMethod (支付方式)

```typescript
interface PaymentMethod {
  id: string;           // 支付方式ID
  name: string;         // 显示名称
  type: string;         // 支付方式类型
  available: boolean;   // 是否可用
}
```

### PackageInfoRequest (套餐信息请求)

```typescript
interface PackageInfoRequest {
  packageId: number;    // 套餐ID
  currency: string;     // 货币类型
}
```

### PackageInfoResponse (套餐信息响应)

```typescript
interface PackageInfoResponse {
  packageInfo: Package;           // 套餐信息
  paymentMethods: PaymentMethod[]; // 可用支付方式列表
}
```

### CreatePaymentRequest (创建支付请求)

```typescript
interface CreatePaymentRequest {
  packageId: number;    // 套餐ID
  currency: string;     // 货币类型
  paymentMethod: string; // 支付方式
}
```

### CreatePaymentResponse (创建支付响应)

```typescript
interface CreatePaymentResponse {
  orderNo: string;          // 订单号
  paymentRequestId: string; // 支付请求ID
  paymentUrl: string;       // 支付跳转URL

  currency: string;         // 货币类型
  paymentStatus: number;    // 支付状态
  success: boolean;         // 是否成功
  paymentMethod: string;    // 支付方式
  message: string;          // 响应消息
  createdTime: string;      // 订单创建时间
  qrCodeUrl?: string;       // 支付二维码URL (可选)

```

### PaymentStatusResponse (支付状态响应)

```typescript
interface PaymentStatusResponse {
  status: number;       // 支付状态 (0=待支付, 1=已支付, 2=已取消, 3=已退款)

```

## API调用方法

### 1. 获取套餐信息和支付方式

```typescript
import { api } from '@/lib/api';

// 调用方法
const getPackageInfo = async (packageId: number, currency: string) => {
  try {
    const response = await api.package.getPackageInfo({
      packageId,
      currency
    });
    
    if (response.code === 200) {
      const { packageInfo, paymentMethods } = response.data;
      console.log('套餐信息:', packageInfo);
      console.log('支付方式:', paymentMethods);
      return response.data;
    } else {
      throw new Error(response.message);
    }
  } catch (error) {
    console.error('获取套餐信息失败:', error);
    throw error;
  }
};

// 使用示例
const packageData = await getPackageInfo(1, 'CNY');
```

### 2. 创建支付订单

```typescript
import { api } from '@/lib/api';

// 调用方法
const createPayment = async (
  packageId: number,
  currency: string,
  paymentMethod: string
) => {
  try {
    const response = await api.package.createPayment({
      packageId,
      currency,
      paymentMethod
    });
    
    if (response.code === 200 && response.data) {
      console.log('支付订单创建成功:', response.data);
      return response.data;
    } else {
      throw new Error(response.message || 'Failed to create payment');
    }
  } catch (error) {
    console.error('创建支付订单失败:', error);
    throw error;
  }
};

// 使用示例
const paymentData = await createPayment(1, 'CNY', 'alipay');
```

## React Hook 使用示例

### 支付流程完整示例

```typescript
import { useState, useEffect } from 'react';
import { api, Package, PaymentMethod, CreatePaymentResponse } from '@/lib/api';
import { toast } from 'sonner';

const usePaymentFlow = (packageId: number, currency: string) => {
  const [packageInfo, setPackageInfo] = useState<Package | null>(null);
  const [paymentMethods, setPaymentMethods] = useState<PaymentMethod[]>([]);
  const [selectedPaymentMethod, setSelectedPaymentMethod] = useState<string>('');
  const [paymentData, setPaymentData] = useState<CreatePaymentResponse | null>(null);
  const [isLoading, setIsLoading] = useState(false);
  const [paymentStep, setPaymentStep] = useState<'loading' | 'select-payment' | 'payment-pending' | 'payment-success' | 'payment-failed'>('loading');

  // 获取套餐信息
  useEffect(() => {
    const fetchPackageInfo = async () => {
      try {
        setIsLoading(true);
        const response = await api.package.getPackageInfo({
          packageId,
          currency
        });

        if (response.code === 200) {
          setPackageInfo(response.data.packageInfo);
          setPaymentMethods(response.data.paymentMethods);
          setPaymentStep('select-payment');
        } else {
          throw new Error(response.message);
        }
      } catch (error) {
        console.error('Failed to fetch package info:', error);
        toast.error('Failed to load package information');
        setPaymentStep('payment-failed');
      } finally {
        setIsLoading(false);
      }
    };

    fetchPackageInfo();
  }, [packageId, currency]);

  // 创建支付订单
  const handleCreatePayment = async () => {
    if (!selectedPaymentMethod || !packageInfo) return;

    try {
      setIsLoading(true);
      const response = await api.package.createPayment({
        packageId,
        currency,
        paymentMethod: selectedPaymentMethod
      });

      if (response.code === 200 && response.data) {
        setPaymentData(response.data);
        setPaymentStep('payment-pending');
        
        // 开始轮询支付状态
        startPaymentStatusPolling(response.data.orderNo);
      } else {
        throw new Error(response.message || 'Failed to create payment');
      }
    } catch (error) {
      console.error('Failed to create payment:', error);
      toast.error('Failed to create payment order');
      setPaymentStep('payment-failed');
    } finally {
      setIsLoading(false);
    }
  };

  // 轮询支付状态
  const startPaymentStatusPolling = (orderNo: string) => {
    const pollInterval = setInterval(async () => {
      try {
        const response = await api.package.getPaymentStatus(orderNo);
        
        if (response.code === 200 && response.data) {
          const { status } = response.data;
          
          if (status === 1) { // 支付成功
            clearInterval(pollInterval);
            setPaymentStep('payment-success');
            toast.success('Payment successful!');
          } else if (status === 2 || status === 3) { // 已取消或已退款
            clearInterval(pollInterval);
            setPaymentStep('payment-failed');
            toast.error('Payment failed or cancelled');
          }
        }
      } catch (error) {
        console.error('Failed to check payment status:', error);
      }
    }, 3000); // 每3秒检查一次

    // 5分钟后停止轮询
    setTimeout(() => {
      clearInterval(pollInterval);
    }, 300000);
  };

  return {
    packageInfo,
    paymentMethods,
    selectedPaymentMethod,
    setSelectedPaymentMethod,
    paymentData,
    paymentStep,
    isLoading,
    handleCreatePayment
  };
};

export default usePaymentFlow;
```

## 支付状态说明

| 状态码 | 状态名称 | 说明 |
|--------|----------|------|
| 0      | 待支付   | 订单已创建，等待用户支付 |
| 1      | 已支付   | 支付成功，订单完成 |
| 2      | 已取消   | 订单已取消 |
| 3      | 已退款   | 订单已退款 |

## 错误处理

```typescript
// 统一错误处理示例
const handlePaymentError = (error: any) => {
  if (error.response?.data?.code === 4001) {
    toast.error('重复的支付请求，请勿重复提交');
  } else if (error.response?.data?.code === 4002) {
    toast.error('套餐不存在');
  } else if (error.response?.data?.code === 4003) {
    toast.error('套餐已下架');
  } else {
    toast.error('支付处理失败，请稍后重试');
  }
};
```

## 注意事项

1. 所有API调用都会自动处理JWT Token认证
2. 支付金额字段为number类型，前端显示时注意精度处理
3. 时间字段使用ISO 8601格式字符串
4. 建议使用轮询方式检查支付状态，避免用户长时间等待
5. 支付二维码URL可能为空，使用前需要检查
6. 货币类型支持实时汇率转换，价格会根据选择的货币动态计算
