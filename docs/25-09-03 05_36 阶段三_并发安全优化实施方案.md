#### 3.1.2 Redis分布式锁实现
```java
@Component
@Slf4j
public class RedisDistributedLock implements DistributedLock {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    private static final String LOCK_PREFIX = "distributed_lock:";
    private static final String UNLOCK_SCRIPT = 
        "if redis.call('get', KEYS[1]) == ARGV[1] then " +
        "return redis.call('del', KEYS[1]) " +
        "else return 0 end";
    
    @Override
    public boolean tryLock(String key, String value, long expireTime) {
        try {
            String lockKey = LOCK_PREFIX + key;
            Boolean result = redisTemplate.opsForValue()
                .setIfAbsent(lockKey, value, Duration.ofMillis(expireTime));
            
            if (Boolean.TRUE.equals(result)) {
                LogUtils.logDebug(log, "分布式锁", "获取锁成功: {}", lockKey);
                return true;
            } else {
                LogUtils.logDebug(log, "分布式锁", "获取锁失败: {}", lockKey);
                return false;
            }
        } catch (Exception e) {
            LogUtils.logError(log, "分布式锁", "获取锁异常: {}", e, key);
            return false;
        }
    }
    
    @Override
    public boolean releaseLock(String key, String value) {
        try {
            String lockKey = LOCK_PREFIX + key;
            Long result = redisTemplate.execute((RedisCallback<Long>) connection -> {
                return connection.eval(UNLOCK_SCRIPT.getBytes(),
                    RedisClusterConnection.ReturnType.INTEGER, 1,
                    lockKey.getBytes(), value.getBytes());
            });
            
            boolean released = result != null && result > 0;
            if (released) {
                LogUtils.logDebug(log, "分布式锁", "释放锁成功: {}", lockKey);
            } else {
                LogUtils.logDebug(log, "分布式锁", "释放锁失败或锁已过期: {}", lockKey);
            }
            
            return released;
        } catch (Exception e) {
            LogUtils.logError(log, "分布式锁", "释放锁异常: {}", e, key);
            return false;
        }
    }
    
    @Override
    public <T> T executeWithLock(String key, long waitTime, long expireTime, Supplier<T> supplier) {
        String lockValue = UUID.randomUUID().toString();
        long startTime = System.currentTimeMillis();
        
        while (System.currentTimeMillis() - startTime < waitTime) {
            if (tryLock(key, lockValue, expireTime)) {
                try {
                    return supplier.get();
                } finally {
                    releaseLock(key, lockValue);
                }
            }
            
            // 短暂等待后重试
            try {
                Thread.sleep(10);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                throw new RuntimeException("Lock acquisition interrupted", e);
            }
        }
        
        throw new RuntimeException("Failed to acquire lock within timeout: " + key);
    }
}
```

#### 3.1.3 分布式锁模板工具
```java
@Component
@Slf4j
public class DistributedLockTemplate {
    
    @Autowired
    private DistributedLock distributedLock;
    
    /**
     * 执行带锁的操作
     */
    public <T> T execute(String lockKey, long timeoutMs, Supplier<T> supplier) {
        return distributedLock.executeWithLock(lockKey, timeoutMs, timeoutMs, supplier);
    }
    
    /**
     * 执行带锁的操作（无返回值）
     */
    public void execute(String lockKey, long timeoutMs, Runnable runnable) {
        distributedLock.executeWithLock(lockKey, timeoutMs, timeoutMs, () -> {
            runnable.run();
            return null;
        });
    }
    
    /**
     * 用户级别的锁操作
     */
    public <T> T executeUserLock(Long userId, String operation, long timeoutMs, Supplier<T> supplier) {
        String lockKey = String.format("user:%d:%s", userId, operation);
        return execute(lockKey, timeoutMs, supplier);
    }
}
```

### 3.2 第二步：优化UserAccountServiceImpl并发安全（2-3天）

#### 3.2.1 重构updateUsage方法
```java
@Service
@Slf4j
public class UserAccountServiceImpl extends ServiceImpl<UserAccountDao, UserAccount>
        implements UserAccountService {

    @Autowired
    private DistributedLockTemplate distributedLockTemplate;
    
    @Autowired
    private HybridCacheManager hybridCacheManager;

    @Override
    @Transactional
    public boolean updateUsage(Long userId, Integer usedCount) {
        if (userId == null || usedCount == null || usedCount <= 0) {
            return false;
        }

        // 使用分布式锁确保并发安全
        return distributedLockTemplate.executeUserLock(userId, "update_usage", 3000, () -> {
            return performAtomicUsageUpdate(userId, usedCount);
        });
    }

    private boolean performAtomicUsageUpdate(Long userId, Integer usedCount) {
        try {
            LocalDateTime updateTime = getBusinessDateTime();
            
            // 使用数据库原子操作更新剩余次数
            int updatedRows = baseMapper.decrementChatCountAtomic(userId, usedCount, updateTime);
            
            if (updatedRows > 0) {
                // 更新总使用次数
                baseMapper.incrementTotalChatCount(userId, usedCount, updateTime);
                
                // 清除相关缓存
                evictUserAccountCache(userId);
                
                LogUtils.logUserDebug(log, userId, "更新使用次数", 
                    "原子更新成功，使用次数: {}", usedCount);
                return true;
            } else {
                LogUtils.logUserWarning(log, userId, "更新使用次数", 
                    "原子更新失败，可能余额不足或套餐已过期，使用次数: {}", usedCount);
                return false;
            }

        } catch (Exception e) {
            LogUtils.logUserError(log, userId, "更新使用次数", 
                "原子更新异常，使用次数: {}", e, usedCount);
            return false;
        }
    }

    /**
     * 批量更新使用次数（支持多个用户同时更新）
     */
    @Transactional
    public Map<Long, Boolean> batchUpdateUsage(Map<Long, Integer> userUsageMap) {
        Map<Long, Boolean> results = new ConcurrentHashMap<>();
        
        // 按用户ID排序，避免死锁
        List<Long> sortedUserIds = userUsageMap.keySet().stream()
            .sorted()
            .collect(Collectors.toList());
        
        for (Long userId : sortedUserIds) {
            Integer usedCount = userUsageMap.get(userId);
            boolean result = updateUsage(userId, usedCount);
            results.put(userId, result);
        }
        
        return results;
    }

    private void evictUserAccountCache(Long userId) {
        String cacheKey = "user:account:" + userId;
        hybridCacheManager.evict(cacheKey);
    }
}
```

```sql
<update id="decrementChatCountAtomic">
    UPDATE user_account 
    SET chat_count_remaining = GREATEST(
            CASE 
                WHEN chat_count_remaining IS NULL THEN -#{usedCount}
                ELSE chat_count_remaining - #{usedCount} 
            END, 
            0
        ),

        version = version + 1
```

```java
@Component
@Slf4j
public class ConcurrencyMonitor {
    
    private final MeterRegistry meterRegistry;
    private final Timer lockAcquisitionTimer;
    private final Counter lockFailureCounter;
    private final Counter concurrentUpdateCounter;
    
    public ConcurrencyMonitor(MeterRegistry meterRegistry) {
        this.meterRegistry = meterRegistry;
        this.lockAcquisitionTimer = Timer.builder("lock.acquisition.time")
            .description("Time to acquire distributed lock")
            .register(meterRegistry);
        this.lockFailureCounter = Counter.builder("lock.acquisition.failure")
            .description("Failed lock acquisitions")
            .register(meterRegistry);
        this.concurrentUpdateCounter = Counter.builder("concurrent.update")
            .description("Concurrent update operations")
            .register(meterRegistry);
    }
    
    public void recordLockAcquisition(long timeMs, boolean success) {
        lockAcquisitionTimer.record(timeMs, TimeUnit.MILLISECONDS);
        if (!success) {
            lockFailureCounter.increment();
        }
    }
    
    public void recordConcurrentUpdate(String operation) {
        concurrentUpdateCounter.increment(Tags.of("operation", operation));
    }
    
    @EventListener
    @Scheduled(fixedRate = 60000) // 每分钟检查一次
    public void checkConcurrencyHealth() {
        double lockFailureRate = lockFailureCounter.count() / 
            (lockAcquisitionTimer.count() > 0 ? lockAcquisitionTimer.count() : 1);
        
        if (lockFailureRate > 0.1) { // 失败率超过10%
            log.warn("分布式锁失败率过高: {}%", lockFailureRate * 100);
            // 发送告警...
        }
        
        double avgLockTime = lockAcquisitionTimer.mean(TimeUnit.MILLISECONDS);
        if (avgLockTime > 1000) { // 平均获取锁时间超过1秒
            log.warn("分布式锁获取时间过长: {}ms", avgLockTime);
            // 发送告警...
        }
    }
}
```

```java
@Component
@Slf4j
public class DeadlockDetector {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    /**
     * 检测并清理过期锁
     */
    @Scheduled(fixedRate = 30000) // 每30秒检查一次
    public void detectAndCleanExpiredLocks() {
        try {
            Set<String> lockKeys = redisTemplate.keys("distributed_lock:*");
            if (lockKeys == null || lockKeys.isEmpty()) {
                return;
            }
            
            int cleanedCount = 0;
            for (String lockKey : lockKeys) {
                Long ttl = redisTemplate.getExpire(lockKey);
                if (ttl != null && ttl < 0) { // 永久键，可能是异常情况
                    redisTemplate.delete(lockKey);
                    cleanedCount++;
                    log.warn("清理异常分布式锁: {}", lockKey);
                }
            }
            
            if (cleanedCount > 0) {
                log.info("清理了{}个异常分布式锁", cleanedCount);
            }
            
        } catch (Exception e) {
            log.error("检测分布式锁异常", e);
        }
    }
    
    /**
     * 强制释放用户锁（紧急情况使用）
     */
    public boolean forceReleaseLock(Long userId, String operation) {
        try {
            String lockKey = String.format("distributed_lock:user:%d:%s", userId, operation);
            Boolean deleted = redisTemplate.delete(lockKey);
            
            if (Boolean.TRUE.equals(deleted)) {
                log.warn("强制释放用户锁: userId={}, operation={}", userId, operation);
                return true;
            }
            
            return false;
        } catch (Exception e) {
            log.error("强制释放锁异常: userId={}, operation={}", userId, operation, e);
            return false;
        }
    }
}
```

```java
@Component
@Slf4j
public class DataConsistencyChecker {
    
    @Autowired
    private UserAccountService userAccountService;
    
    @Autowired
    private ChatMessageService chatMessageService;
    
    /**
     * 检查用户使用次数一致性

        try {
            List<Long> userIds = userAccountService.getAllActiveUserIds();
            int inconsistentCount = 0;
            
            for (Long userId : userIds) {
                boolean consistent = checkUserUsageConsistency(userId);
                if (!consistent) {
                    inconsistentCount++;
                }
            }
            
            if (inconsistentCount > 0) {
                log.warn("发现{}个用户的使用次数数据不一致", inconsistentCount);
                // 发送告警通知...
            } else {
                log.info("所有用户使用次数数据一致性检查通过");
            }
            
        } catch (Exception e) {
            log.error("使用次数一致性检查异常", e);
        }
    }
    
    private boolean checkUserUsageConsistency(Long userId) {
        try {
            // 从数据库获取用户账户信息
            UserAccount account = userAccountService.getUserAccount(userId);
            if (account == null) {
                return true; // 用户不存在，跳过检查
            }
            
            // 计算实际使用次数
            int actualUsageCount = chatMessageService.countUserMessages(userId);
            int recordedUsageCount = account.getTotalChatCount() != null ? 
                account.getTotalChatCount() : 0;
            
            if (Math.abs(actualUsageCount - recordedUsageCount) > 5) { // 允许5次误差
                log.warn("用户{}使用次数不一致: 实际={}, 记录={}", 
                    userId, actualUsageCount, recordedUsageCount);
                return false;
            }
            
            return true;
            
        } catch (Exception e) {
            log.error("检查用户{}使用次数一致性异常", userId, e);
            return false;
        }
    }
}
```
