### Mock OAuth2服务器配置
```java
@TestConfiguration
public class MockOAuth2TestConfig {

    @MockBean
    private OAuth2Service oauth2Service;

    @PostConstruct
    public void setupMockResponses() {
        // Mock Google用户信息
        OAuth2UserInfo googleUser = OAuth2UserInfo.builder()
            .id("google_test_id_123")
            .email("test@gmail.com")
            .name("Test User")
            .avatar("https://example.com/avatar.jpg")
            .provider("google")
            .build();

        // Mock GitHub用户信息
        OAuth2UserInfo githubUser = OAuth2UserInfo.builder()
            .id("github_test_id_456")
            .email("test@github.com")
            .name("GitHub Test User")
            .avatar("https://github.com/avatar.jpg")
            .provider("github")
            .build();

        // 配置Mock行为
        when(oauth2Service.getUserInfo("google", "valid_google_code"))
            .thenReturn(googleUser);
        when(oauth2Service.getUserInfo("github", "valid_github_code"))
            .thenReturn(githubUser);
        when(oauth2Service.getUserInfo(anyString(), eq("invalid_code")))
            .thenThrow(new OAuth2Exception("INVALID_CODE", "Invalid authorization code"));

        // Mock登录处理
        when(oauth2Service.processLogin(any(OAuth2UserInfo.class), anyString(), anyString()))
            .thenAnswer(invocation -> {
                OAuth2UserInfo userInfo = invocation.getArgument(0);
                // 创建模拟用户和登录结果
                User mockUser = createMockUser(userInfo);
                return OAuth2LoginResult.newUser(mockUser);
            });

        // Mock支持的提供商
        when(oauth2Service.getSupportedProviders())
            .thenReturn(Arrays.asList("google", "github"));
        when(oauth2Service.isProviderSupported("google")).thenReturn(true);
        when(oauth2Service.isProviderSupported("github")).thenReturn(true);
        when(oauth2Service.isProviderSupported("invalid")).thenReturn(false);
    }

    private User createMockUser(OAuth2UserInfo userInfo) {
        User user = new User();
        user.setId(System.currentTimeMillis()); // 模拟ID
        user.setEmail(userInfo.getEmail());
        user.setUsername(userInfo.getName());
        user.setAvatar(userInfo.getAvatar());
        user.setRole("user");
        user.setStatus(1);
        return user;
    }
}
```
