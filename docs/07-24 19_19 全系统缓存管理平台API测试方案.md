### 3. 性能压力测试

```python
def test_performance_stress(self):
    """性能压力测试"""
    def concurrent_request():
        start_time = time.time()
        response = self.get("/stats")
        end_time = time.time()
        return {
            "status_code": response.status_code,
            "response_time": end_time - start_time,
            "timestamp": datetime.now()
        }
    
    # 并发测试
    with ThreadPoolExecutor(max_workers=10) as executor:
        futures = [executor.submit(concurrent_request) 
                  for _ in range(100)]
        results = [future.result() for future in futures]
    
    # 性能分析
    avg_response_time = sum(r["response_time"] for r in results) / len(results)
    success_rate = sum(1 for r in results if r["status_code"] == 200) / len(results)
    
    assert avg_response_time < 1.0  # 平均响应时间小于1秒
    assert success_rate > 0.95      # 成功率大于95%
```

## 🖥️ 终端输出监控

### 监控脚本设计
```python
class TerminalMonitor:
    def __init__(self):
        self.log_patterns = {
            "cache_hit": r"L1缓存命中|L2缓存命中|缓存命中",
            "cache_miss": r"缓存未命中|从数据库加载",
            "cache_clear": r"清空.*缓存|缓存清理",
            "circuit_breaker": r"熔断器.*状态|Redis熔断",
            "performance": r"性能统计|响应时间|QPS",
            "error": r"ERROR|Exception|失败",
            "security": r"访问控制|IP检查|403"
        }
        self.captured_logs = []
        
    def start_monitoring(self):
        """开始监控终端输出"""
        def monitor_process():
            process = subprocess.Popen(
                ["tail", "-f", "logs/application.log"],
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                text=True
            )
            
            for line in process.stdout:
                self.analyze_log_line(line)
                self.captured_logs.append({
                    "timestamp": datetime.now(),
                    "content": line.strip(),
                    "patterns": self.match_patterns(line)
                })
        
        monitor_thread = threading.Thread(target=monitor_process)
        monitor_thread.daemon = True
        monitor_thread.start()
```

### 日志分析功能
```python
def analyze_log_line(self, line: str):
    """分析日志行"""
    patterns = self.match_patterns(line)
    
    if "error" in patterns:
        self.handle_error_log(line)
    elif "performance" in patterns:
        self.extract_performance_metrics(line)
    elif "cache_hit" in patterns:
        self.track_cache_performance(line, "hit")
    elif "cache_miss" in patterns:
        self.track_cache_performance(line, "miss")
```
