### 步骤2: 修改detectContentAndGetResult方法（10分钟）

```java
private ContentDetectionResult detectContentAndGetResult(String content) {
    if (content == null) {
        return new ContentDetectionResult("", false, false);
    }

    // 🔥 原始传输模式：跳过内容处理
    if (ENABLE_RAW_TRANSMISSION) {
        // 仍需进行安全检查（除非明确跳过）
        if (!RAW_MODE_SKIP_FILTERING) {
            if (shouldRestartChat(content)) {
                log.warn("原始模式检测到需要重试的错误消息，触发重试");
                return new ContentDetectionResult("", false, true);
            }
            
            if (shouldFilterContent(content)) {
                log.info("原始模式检测到需要过滤的AI响应内容，已阻止向前端发送");
                return new ContentDetectionResult("", true, false);
            }
        }
        
        // 原始模式：完全不修改内容
        log.debug("原始传输模式：跳过所有内容清理，保持100%原始内容");
        return new ContentDetectionResult(content, false, false);
    }

    // 现有逻辑保持不变
    if (shouldRestartChat(content)) {
        log.warn("检测到需要重试的错误消息，触发重试");
        return new ContentDetectionResult("", false, true);
    }

    if (shouldFilterContent(content)) {
        log.info("检测到需要过滤的AI响应内容，已阻止向前端发送");
        return new ContentDetectionResult("", true, false);
    }

    // 现有的清理逻辑...
    String cleaned = content.replaceAll("[\uFFFD\u0000]", "");
    
    // 现有的缩进处理逻辑...
    String[] lines = cleaned.split("\n", -1);
    StringBuilder result = new StringBuilder();
    
    for (int i = 0; i < lines.length; i++) {
        String line = lines[i];
        if (!line.trim().isEmpty()) {
            int leadingSpaces = 0;
            while (leadingSpaces < line.length() && line.charAt(leadingSpaces) == ' ') {
                leadingSpaces++;
            }
            
            String indent = line.substring(0, leadingSpaces);
            String content_part = line.substring(leadingSpaces);
            String cleanedContent = content_part.replaceAll(" {2,}", " ");
            
            result.append(indent).append(cleanedContent);
        } else {
            result.append(line);
        }
        
        if (i < lines.length - 1) {
            result.append("\n");
        }
    }
    
    cleaned = result.toString();
    return new ContentDetectionResult(cleaned, false, false);
}
```

### 步骤3: 修改智能分块发送（10分钟）

```java
private void sendContentBySmartChunking(String content, HttpServletResponse response, ObjectMapper mapper) {
    try {
        int optimalChunkSize = getOptimalChunkSize(content);
        int[] codePoints = content.codePoints().toArray();
        StringBuilder chunk = new StringBuilder();
        int sentChars = 0;

        for (int i = 0; i < codePoints.length; i++) {
            int codePoint = codePoints[i];
            String character = new String(Character.toChars(codePoint));

            // 🔥 原始模式：不过滤任何字符
            if (ENABLE_RAW_TRANSMISSION && RAW_MODE_PRESERVE_ALL_CHARS) {
                chunk.append(character);
            } else {
                // 现有逻辑：过滤无效字符
                if (codePoint != 0xFFFD && codePoint != 0x0000) {
                    chunk.append(character);
                }
            }

            if (chunk.length() >= optimalChunkSize || i == codePoints.length - 1) {
                try {
                    String chunkContent = chunk.toString();
                    String vercelFormat = String.format("0:%s\n", mapper.writeValueAsString(chunkContent));
                    response.getWriter().write(vercelFormat);
                    response.getWriter().flush();

                    sentChars += chunkContent.length();
                    chunk.setLength(0);

                    if (CHUNK_DELAY_MS > 0) {
                        Thread.sleep(CHUNK_DELAY_MS);
                    }
                } catch (InterruptedException e) {
                    log.warn("分块发送被中断: {}", e.getMessage());
                    Thread.currentThread().interrupt();
                } catch (IOException e) {
                    log.error("发送分块内容失败: {}", e.getMessage());
                    chunk.setLength(0);
                }
            }
        }

        log.debug("智能分块发送完成，原始模式: {}, 总字符数: {}, 实际发送: {}", 
                 ENABLE_RAW_TRANSMISSION, codePoints.length, sentChars);

    } catch (Exception e) {
        log.error("智能分块发送失败，回退到块发送: {}", e.getMessage());
        try {
            String vercelFormat = String.format("0:%s\n", mapper.writeValueAsString(content));
            response.getWriter().write(vercelFormat);
            response.getWriter().flush();
        } catch (Exception fallbackException) {
            log.error("回退发送也失败: {}", fallbackException.getMessage());
        }
    }
}
```
