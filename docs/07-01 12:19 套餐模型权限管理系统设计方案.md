# 套餐模型权限管理系统设计方案

**📅 设计时间**: 2025-07-01 12:19:00  
**👤 设计人员**: Augment Agent  
**📊 文档版本**: v1.0  
**🎯 设计目标**: 实现基于套餐类型的AI模型使用权限控制

## 📋 需求分析

### 业务需求
实现基于套餐类型的模型使用权限控制，不同套餐对不同模型有不同的每日使用限制：

| 套餐类型 | STANDARD | PERFORMANCE | BETA MODEL |
|---------|----------|-------------|------------|
| basic   | 300次/天 | 20次/天     | 5次/天     |
| medium  | 500次/天 | 300次/天    | 20次/天    |
| high    | 无限制   | 无限制      | 无限制     |

### 技术需求
- 灵活的权限配置管理
- 高性能的使用次数检查
- 准确的每日使用统计
- 与现有系统的无缝集成

## 🎯 设计方案

### 方案选择：扩展现有每日使用表（推荐）

基于现有的`UserDailyUsage`机制，扩展支持模型维度的使用限制管理。

#### 优势分析
- ✅ 复用现有的每日使用管理机制
- ✅ 保持数据一致性和时区处理逻辑
- ✅ 支持灵活的权限配置
- ✅ 便于监控和统计分析
- ✅ 与现有系统集成度高

```sql
  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键ID',

  `create_time` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_user_model_date` (`user_id`, `model_name`, `usage_date`),
  KEY `idx_user_date` (`user_id`, `usage_date`),
  KEY `idx_package_model` (`package_id`, `model_name`),
  KEY `idx_usage_date` (`usage_date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci 
COMMENT='用户模型每日使用记录表';
```

```java
package com.instser.chat.service.serviceimpl;

import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.instser.chat.dao.PackageModelLimitDao;
import com.instser.chat.dao.UserModelDailyUsageDao;
import com.instser.chat.entity.Package;
import com.instser.chat.entity.PackageModelLimit;
import com.instser.chat.entity.UserAccount;
import com.instser.chat.entity.UserModelDailyUsage;
import com.instser.chat.service.ModelUsageLimitService;
import com.instser.chat.service.PackageService;
import com.instser.chat.service.UserAccountService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.ZoneId;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

/**
 * 模型使用限制服务实现类
 *
 * @author instser
 * @since 2025-07-01
 */
@Slf4j
@Service
public class ModelUsageLimitServiceImpl extends ServiceImpl<UserModelDailyUsageDao, UserModelDailyUsage>
        implements ModelUsageLimitService {

    /**
     * 业务时区 - 统一使用上海时区
     */
    private static final ZoneId BUSINESS_ZONE = ZoneId.of("Asia/Shanghai");

    @Autowired
    private PackageModelLimitDao packageModelLimitDao;

    @Autowired
    private UserAccountService userAccountService;

    @Autowired
    private PackageService packageService;

    /**
     * 获取业务日期（上海时区）
     */
    private LocalDate getBusinessDate() {
        return LocalDate.now(BUSINESS_ZONE);
    }

    /**
     * 获取业务时间（上海时区）
     */
    private LocalDateTime getBusinessDateTime() {
        return LocalDateTime.now(BUSINESS_ZONE);
    }

    @Override
    public boolean canUseModel(Long userId, String modelName) {
        try {
            // 1. 获取用户账户信息
            UserAccount userAccount = userAccountService.getUserAccount(userId);
            if (userAccount == null || userAccount.getPackageId() == null) {
                log.warn("用户{}没有有效套餐", userId);
                return false;
            }

            // 2. 检查套餐是否过期
            if (userAccount.getPackageExpireTime() != null &&
                userAccount.getPackageExpireTime().isBefore(getBusinessDateTime())) {
                log.warn("用户{}的套餐已过期", userId);
                return false;
            }

            // 3. 获取套餐信息
            Package packageInfo = packageService.getPackageById(userAccount.getPackageId());
            if (packageInfo == null) {
                log.warn("用户{}的套餐信息不存在，套餐ID：{}", userId, userAccount.getPackageId());
                return false;
            }

            // 4. 获取模型限制配置
            PackageModelLimit modelLimit = packageModelLimitDao.selectByPackageTypeAndModel(
                packageInfo.getPackageType(), modelName);

            if (modelLimit == null || modelLimit.getIsEnabled() != 1) {
                log.warn("套餐类型{}不支持模型{}或配置已禁用", packageInfo.getPackageType(), modelName);
                return false;
            }

            // 5. 如果是无限制套餐，直接允许
            if (modelLimit.getDailyLimit() == -1) {
                return true;
            }

            // 6. 检查今日使用情况
            UserModelDailyUsage todayUsage = getTodayUsage(userId, modelName);
            if (todayUsage == null) {
                return true; // 今日未使用，可以使用
            }

            return todayUsage.getUsedCount() < todayUsage.getDailyLimit();

        } catch (Exception e) {
            log.error("检查用户{}模型{}使用权限异常", userId, modelName, e);
            return false;
        }
    }

    @Override
    @Transactional
    public boolean recordModelUsage(Long userId, String modelName) {
        try {
            // 1. 获取用户账户信息
            UserAccount userAccount = userAccountService.getUserAccount(userId);
            if (userAccount == null || userAccount.getPackageId() == null) {
                log.error("用户{}没有有效套餐，无法记录模型使用", userId);
                return false;
            }

            // 2. 获取套餐信息
            Package packageInfo = packageService.getPackageById(userAccount.getPackageId());
            if (packageInfo == null) {
                log.error("用户{}的套餐信息不存在，套餐ID：{}", userId, userAccount.getPackageId());
                return false;
            }

            // 3. 获取模型限制配置
            PackageModelLimit modelLimit = packageModelLimitDao.selectByPackageTypeAndModel(
                packageInfo.getPackageType(), modelName);

            if (modelLimit == null || modelLimit.getIsEnabled() != 1) {
                log.error("套餐类型{}不支持模型{}或配置已禁用", packageInfo.getPackageType(), modelName);
                return false;
            }

            // 4. 如果是无限制套餐，记录但不限制
            if (modelLimit.getDailyLimit() == -1) {
                // 仍然记录使用情况，用于统计
                baseMapper.insertOrIncrementUsage(userId, userAccount.getPackageId(),
                    modelName, getBusinessDate(), -1);
                log.info("用户{}使用无限制模型{}，已记录使用", userId, modelName);
                return true;
            }

            // 5. 检查当前使用情况，避免超限
            UserModelDailyUsage todayUsage = getTodayUsage(userId, modelName);
            if (todayUsage != null && todayUsage.getUsedCount() >= todayUsage.getDailyLimit()) {
                log.warn("用户{}今日模型{}使用次数已达到限制：{}/{}",
                        userId, modelName, todayUsage.getUsedCount(), todayUsage.getDailyLimit());
                return false;
            }

            // 6. 使用原子性操作插入或更新使用记录
            int updated = baseMapper.insertOrIncrementUsage(userId, userAccount.getPackageId(),
                    modelName, getBusinessDate(), modelLimit.getDailyLimit());

            if (updated > 0) {
                // 重新获取更新后的记录用于日志
                UserModelDailyUsage updatedUsage = getTodayUsage(userId, modelName);
                if (updatedUsage != null) {
                    log.info("用户{}模型{}使用记录更新成功，当前使用：{}/{}，业务日期：{}",
                            userId, modelName, updatedUsage.getUsedCount(),
                            updatedUsage.getDailyLimit(), getBusinessDate());
                }
                return true;
            }

            log.error("用户{}模型{}使用记录更新失败", userId, modelName);
            return false;

        } catch (Exception e) {
            log.error("记录用户{}模型{}使用异常", userId, modelName, e);
            return false;
        }
    }

    @Override
    public int getRemainingUsage(Long userId, String modelName) {
        try {
            // 1. 获取用户账户信息
            UserAccount userAccount = userAccountService.getUserAccount(userId);
            if (userAccount == null || userAccount.getPackageId() == null) {
                return 0;
            }

            // 2. 获取套餐信息
            Package packageInfo = packageService.getPackageById(userAccount.getPackageId());
            if (packageInfo == null) {
                return 0;
            }

            // 3. 获取模型限制配置
            PackageModelLimit modelLimit = packageModelLimitDao.selectByPackageTypeAndModel(
                packageInfo.getPackageType(), modelName);

            if (modelLimit == null || modelLimit.getIsEnabled() != 1) {
                return 0;
            }

            // 4. 如果是无限制套餐
            if (modelLimit.getDailyLimit() == -1) {
                return -1; // 表示无限制
            }

            // 5. 获取今日使用情况
            UserModelDailyUsage todayUsage = getTodayUsage(userId, modelName);
            if (todayUsage == null) {
                return modelLimit.getDailyLimit();
            }

            return Math.max(0, todayUsage.getDailyLimit() - todayUsage.getUsedCount());

        } catch (Exception e) {
            log.error("获取用户{}模型{}剩余使用次数异常", userId, modelName, e);
            return 0;
        }
    }

    @Override
    public UserModelDailyUsage getTodayUsage(Long userId, String modelName) {
        return baseMapper.selectByUserModelDate(userId, modelName, getBusinessDate());
    }

    @Override
    public Map<String, Integer> getPackageModelLimits(String packageType) {
        try {
            List<PackageModelLimit> limits = packageModelLimitDao.selectByPackageType(packageType);
            return limits.stream()
                    .filter(limit -> limit.getIsEnabled() == 1)
                    .collect(Collectors.toMap(
                            PackageModelLimit::getModelName,
                            PackageModelLimit::getDailyLimit
                    ));
        } catch (Exception e) {
            log.error("获取套餐类型{}的模型限制配置异常", packageType, e);
            return new HashMap<>();
        }
    }

    @Override
    public List<UserModelDailyUsage> getUserModelUsageToday(Long userId) {
        try {
            return baseMapper.selectRecentUsage(userId, 1);
        } catch (Exception e) {
            log.error("获取用户{}今日模型使用情况异常", userId, e);
            return List.of();
        }
    }

    @Override
    @Transactional
    public void initPackageModelLimits() {
        // 检查是否已有配置数据
        if (packageModelLimitDao.selectCount(null) > 0) {
            log.info("套餐模型限制配置已存在，跳过初始化");
            return;
        }

        log.info("开始初始化套餐模型限制配置");

        // 创建配置数据
        String[][] configs = {
            // 基础套餐配置
            {"basic", "STANDARD", "300", "基础套餐-标准模型每日300次"},
            {"basic", "PERFORMANCE", "20", "基础套餐-性能模型每日20次"},
            {"basic", "BETA_MODEL", "5", "基础套餐-测试模型每日5次"},
            // 中级套餐配置
            {"medium", "STANDARD", "500", "中级套餐-标准模型每日500次"},
            {"medium", "PERFORMANCE", "300", "中级套餐-性能模型每日300次"},
            {"medium", "BETA_MODEL", "20", "中级套餐-测试模型每日20次"},
            // 高级套餐配置
            {"high", "STANDARD", "-1", "高级套餐-标准模型无限制"},
            {"high", "PERFORMANCE", "-1", "高级套餐-性能模型无限制"},
            {"high", "BETA_MODEL", "-1", "高级套餐-测试模型无限制"}
        };

        for (String[] config : configs) {
            PackageModelLimit limit = new PackageModelLimit();
            limit.setPackageType(config[0]);
            limit.setModelName(config[1]);
            limit.setDailyLimit(Integer.parseInt(config[2]));
            limit.setDescription(config[3]);
            limit.setIsEnabled(1);
            limit.setCreateTime(getBusinessDateTime());
            limit.setUpdateTime(getBusinessDateTime());

            packageModelLimitDao.insert(limit);
        }

        log.info("套餐模型限制配置初始化完成");
    }

    @Override
    @Transactional
    public int cleanExpiredUsageRecords(int daysToKeep) {
        try {
            LocalDate beforeDate = getBusinessDate().minusDays(daysToKeep);
            int cleaned = baseMapper.deleteExpiredRecords(beforeDate);
            log.info("清理{}天前的模型使用记录完成，共清理{}条记录", daysToKeep, cleaned);
            return cleaned;
        } catch (Exception e) {
            log.error("清理过期模型使用记录异常", e);
            return 0;
        }
    }
}
```
