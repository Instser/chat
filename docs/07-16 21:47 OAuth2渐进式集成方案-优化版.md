```java
@RestController
@RequestMapping("/auth/oauth2")
@Slf4j
@Validated
public class OAuth2Controller {

    @Autowired
    private OAuth2Service oauth2Service;
    
    @Autowired
    private OAuth2SecurityUtil securityUtil;
    
    @Autowired
    private OAuth2AuditService auditService;
    
    @Autowired
    private JwtUtil jwtUtil;

    /**
     * 获取OAuth2登录URL - 增强版
     */
    @GetMapping("/login/{provider}")
    @RateLimiter(name = "oauth2-login-url")
    public ResponseEntity<Result<OAuth2LoginResponse>> getLoginUrl(
            @PathVariable @Pattern(regexp = "^(google|github)$") String provider,
            HttpServletRequest request) {
        
        long startTime = System.currentTimeMillis();
        String clientIp = getClientIp(request);
        String userAgent = request.getHeader("User-Agent");
        
        try {
            // 1. 生成安全状态令牌
            String state = securityUtil.generateSecureState();
            
            // 2. 存储状态信息
            oauth2Service.saveState(state, provider, clientIp, userAgent);
            
            // 3. 生成授权URL
            String authUrl = oauth2Service.getAuthorizationUrl(provider, state);
            
            // 4. 记录审计日志
            auditService.logOAuth2Event(null, provider, "GET_LOGIN_URL", "SUCCESS", 
                clientIp, userAgent, System.currentTimeMillis() - startTime, null, null);
            
            OAuth2LoginResponse response = OAuth2LoginResponse.builder()
                .authUrl(authUrl)
                .provider(provider)
                .state(state)
                .expiresIn(300) // 5分钟过期
                .build();
                
            return ResponseEntity.ok(Result.success(response));
            
        } catch (Exception e) {
            log.error("获取OAuth2登录URL失败: provider={}, error={}", provider, e.getMessage());
            auditService.logOAuth2Event(null, provider, "GET_LOGIN_URL", "FAILED", 
                clientIp, userAgent, System.currentTimeMillis() - startTime, 
                "URL_GENERATION_ERROR", e.getMessage());
            
            return ResponseEntity.ok(Result.error(OAuth2ErrorCode.URL_GENERATION_FAILED, 
                "获取登录链接失败，请稍后重试"));
        }
    }

    /**
     * OAuth2回调处理 - 增强版
     */
    @GetMapping("/callback/{provider}")
    public ResponseEntity<?> handleCallback(
            @PathVariable @Pattern(regexp = "^(google|github)$") String provider,
            @RequestParam String code,
            @RequestParam String state,
            @RequestParam(required = false) String error,
            HttpServletRequest request,
            HttpServletResponse response) {
        
        long startTime = System.currentTimeMillis();
        String clientIp = getClientIp(request);
        String userAgent = request.getHeader("User-Agent");
        
        try {
            // 1. 处理OAuth错误
            if (StringUtils.hasText(error)) {
                return handleOAuthError(error, provider, clientIp, userAgent);
            }
            
            // 2. 验证状态令牌
            if (!securityUtil.validateState(state, provider, clientIp)) {
                auditService.logOAuth2Event(null, provider, "CALLBACK", "FAILED", 
                    clientIp, userAgent, System.currentTimeMillis() - startTime, 
                    "INVALID_STATE", "状态令牌验证失败");
                return redirectToError("invalid_state", "安全验证失败");
            }
            
            // 3. 获取用户信息
            OAuth2UserInfo userInfo = oauth2Service.getUserInfo(provider, code);
            
            // 4. 处理用户登录/注册
            OAuth2LoginResult loginResult = oauth2Service.processLogin(userInfo, clientIp, userAgent);
            
            // 5. 生成JWT令牌
            String token = jwtUtil.generateToken(loginResult.getUser());
            
            // 6. 设置认证Cookie
            ResponseCookie authCookie = jwtUtil.createAuthCookie(token);
            response.addHeader(HttpHeaders.SET_COOKIE, authCookie.toString());
            
            // 7. 记录成功日志
            auditService.logOAuth2Event(loginResult.getUser().getId(), provider, "LOGIN", "SUCCESS", 
                clientIp, userAgent, System.currentTimeMillis() - startTime, null, null);
            
            // 8. 重定向到成功页面
            String redirectUrl = buildSuccessRedirectUrl(loginResult);
            return ResponseEntity.status(HttpStatus.FOUND)
                    .header(HttpHeaders.LOCATION, redirectUrl)
                    .build();
                    
        } catch (OAuth2Exception e) {
            log.error("OAuth2回调处理失败: provider={}, error={}", provider, e.getMessage());
            auditService.logOAuth2Event(null, provider, "CALLBACK", "FAILED", 
                clientIp, userAgent, System.currentTimeMillis() - startTime, 
                e.getErrorCode(), e.getMessage());
            
            return redirectToError(e.getErrorCode(), e.getMessage());
            
        } catch (Exception e) {
            log.error("OAuth2回调处理异常: provider={}", provider, e);
            auditService.logOAuth2Event(null, provider, "CALLBACK", "ERROR", 
                clientIp, userAgent, System.currentTimeMillis() - startTime, 
                "SYSTEM_ERROR", e.getMessage());
            
            return redirectToError("system_error", "系统错误，请稍后重试");
        }
    }

    /**
     * 绑定OAuth账号 - 新增功能
     */
    @PostMapping("/bind/{provider}")
    @PreAuthorize("hasRole('USER')")
    public ResponseEntity<Result<Void>> bindOAuthAccount(
            @PathVariable String provider,
            @RequestParam String code,
            @RequestParam String state,
            Authentication authentication,
            HttpServletRequest request) {

        String clientIp = getClientIp(request);
        String userAgent = request.getHeader("User-Agent");
        Long userId = getCurrentUserId(authentication);

        try {
            // 验证状态并绑定账号
            oauth2Service.bindOAuthAccount(userId, provider, code, state, clientIp);

            auditService.logOAuth2Event(userId, provider, "BIND", "SUCCESS",
                clientIp, userAgent, 0, null, null);

            return ResponseEntity.ok(Result.success("账号绑定成功"));

        } catch (OAuth2Exception e) {
            auditService.logOAuth2Event(userId, provider, "BIND", "FAILED",
                clientIp, userAgent, 0, e.getErrorCode(), e.getMessage());

            return ResponseEntity.ok(Result.error(e.getErrorCode(), e.getMessage()));
        }
    }

    /**
     * 解绑OAuth账号 - 新增功能
     */
    @DeleteMapping("/unbind/{provider}")
    @PreAuthorize("hasRole('USER')")
    public ResponseEntity<Result<Void>> unbindOAuthAccount(

            Authentication authentication,

        String clientIp = getClientIp(request);
        String userAgent = request.getHeader("User-Agent");
        Long userId = getCurrentUserId(authentication);

        try {
            oauth2Service.unbindOAuthAccount(userId, provider);

            auditService.logOAuth2Event(userId, provider, "UNBIND", "SUCCESS",
                clientIp, userAgent, 0, null, null);

            return ResponseEntity.ok(Result.success("账号解绑成功"));

        } catch (OAuth2Exception e) {
            auditService.logOAuth2Event(userId, provider, "UNBIND", "FAILED",

            return ResponseEntity.ok(Result.error(e.getErrorCode(), e.getMessage()));
        }
    }

    /**
     * 获取用户绑定的OAuth账号列表
     */
    @GetMapping("/bindings")
    @PreAuthorize("hasRole('USER')")
    public ResponseEntity<Result<List<OAuth2BindingInfo>>> getOAuthBindings(
            Authentication authentication) {

        Long userId = getCurrentUserId(authentication);
        List<OAuth2BindingInfo> bindings = oauth2Service.getUserOAuthBindings(userId);

        return ResponseEntity.ok(Result.success(bindings));
    }

    // 私有辅助方法
    private ResponseEntity<?> handleOAuthError(String error, String provider,
                                             String clientIp, String userAgent) {
        String errorMessage = switch (error) {
            case "access_denied" -> "用户拒绝授权";
            case "invalid_request" -> "请求参数无效";
            case "unauthorized_client" -> "客户端未授权";
            case "unsupported_response_type" -> "不支持的响应类型";
            case "invalid_scope" -> "无效的权限范围";
            case "server_error" -> "服务器错误";
            case "temporarily_unavailable" -> "服务暂时不可用";
            default -> "未知错误: " + error;
        };

        auditService.logOAuth2Event(null, provider, "CALLBACK", "OAUTH_ERROR",
            clientIp, userAgent, 0, error, errorMessage);

        return redirectToError(error, errorMessage);
    }

    private ResponseEntity<?> redirectToError(String errorCode, String errorMessage) {
        String errorUrl = oauth2Service.buildErrorRedirectUrl(errorCode, errorMessage);
        return ResponseEntity.status(HttpStatus.FOUND)
                .header(HttpHeaders.LOCATION, errorUrl)
                .build();
    }

    private String buildSuccessRedirectUrl(OAuth2LoginResult loginResult) {
        String baseUrl = oauth2Service.getSuccessRedirectUrl();
        if (loginResult.isNewUser()) {
            return baseUrl + "?welcome=true";
        } else if (loginResult.isNewBinding()) {
            return baseUrl + "?bound=true";
        }
        return baseUrl;
    }

    private String getClientIp(HttpServletRequest request) {
        String xForwardedFor = request.getHeader("X-Forwarded-For");
        if (StringUtils.hasText(xForwardedFor)) {
            return xForwardedFor.split(",")[0].trim();
        }
        String xRealIp = request.getHeader("X-Real-IP");
        if (StringUtils.hasText(xRealIp)) {
            return xRealIp;
        }
        return request.getRemoteAddr();
    }

    private Long getCurrentUserId(Authentication authentication) {
        if (authentication != null && authentication.getPrincipal() instanceof UserDetails) {
            UserDetails userDetails = (UserDetails) authentication.getPrincipal();
            // 假设UserDetails实现类有getId方法
            return ((CustomUserDetails) userDetails).getId();
        }
        throw new OAuth2Exception("UNAUTHORIZED", "用户未登录");
    }
}
```

#### 2. OAuth2服务实现类
```java
@Service
@Slf4j
@Transactional
public class OAuth2ServiceImpl implements OAuth2Service {

    @Autowired
    private UserService userService;

    @Autowired
    private UserOAuthMapper userOAuthMapper;

    @Autowired
    private OAuth2StateMapper oauth2StateMapper;

    @Autowired
    private OAuth2SecurityUtil securityUtil;

    @Autowired
    private OAuth2AuditService auditService;

    @Autowired
    private PasswordEncoder passwordEncoder;

    @Value("${oauth2.redirect-urls.success}")
    private String successRedirectUrl;

    @Value("${oauth2.redirect-urls.error}")
    private String errorRedirectUrl;

    @Override
    public void saveState(String state, String provider, String clientIp, String userAgent) {
        OAuth2State oauth2State = new OAuth2State();
        oauth2State.setStateToken(state);
        oauth2State.setProvider(provider);
        oauth2State.setClientIp(clientIp);
        oauth2State.setUserAgent(userAgent);
        oauth2State.setExpiresAt(LocalDateTime.now().plusMinutes(5)); // 5分钟过期

        oauth2StateMapper.insert(oauth2State);

        log.debug("保存OAuth2状态: state={}, provider={}, ip={}", state, provider, clientIp);
    }

    @Override
    public boolean validateState(String state, String provider, String clientIp) {
        OAuth2State oauth2State = oauth2StateMapper.findByStateToken(state);

        if (oauth2State == null) {
            log.warn("OAuth2状态令牌不存在: state={}", state);
            return false;
        }

        if (oauth2State.getUsed() == 1) {
            log.warn("OAuth2状态令牌已使用: state={}", state);
            return false;
        }

        if (oauth2State.getExpiresAt().isBefore(LocalDateTime.now())) {
            log.warn("OAuth2状态令牌已过期: state={}, expiresAt={}", state, oauth2State.getExpiresAt());
            return false;
        }

        if (!provider.equals(oauth2State.getProvider())) {
            log.warn("OAuth2状态令牌提供商不匹配: expected={}, actual={}", oauth2State.getProvider(), provider);
            return false;
        }

        // IP验证（可选，根据安全需求决定）
        if (!clientIp.equals(oauth2State.getClientIp())) {
            log.warn("OAuth2状态令牌IP不匹配: expected={}, actual={}", oauth2State.getClientIp(), clientIp);
            // 可以选择是否严格验证IP
        }

        // 标记为已使用
        oauth2State.setUsed(1);
        oauth2StateMapper.updateById(oauth2State);

        log.debug("OAuth2状态验证成功: state={}, provider={}", state, provider);
        return true;
    }

    @Override
    public OAuth2LoginResult processLogin(OAuth2UserInfo userInfo, String clientIp, String userAgent) {
        log.info("处理OAuth2登录: provider={}, email={}, ip={}",
            userInfo.getProvider(), userInfo.getEmail(), clientIp);

        // 1. 检查是否已绑定OAuth账号
        UserOAuth existingOAuth = userOAuthMapper.findByProviderAndProviderId(
            userInfo.getProvider(), userInfo.getId());

        if (existingOAuth != null && existingOAuth.getStatus() == 1) {
            // 已绑定且状态正常，更新登录信息
            updateOAuthLoginInfo(existingOAuth, userInfo);
            User user = userService.getById(existingOAuth.getUserId());

            log.info("OAuth2用户登录成功: userId={}, provider={}", user.getId(), userInfo.getProvider());
            return OAuth2LoginResult.existingUser(user);
        }

        // 2. 检查邮箱是否已存在
        User existingUser = userService.getUserByEmail(userInfo.getEmail());
        if (existingUser != null) {
            // 邮箱已存在，绑定OAuth账号
            UserOAuth newOAuth = createOAuthBinding(existingUser.getId(), userInfo);
            userOAuthMapper.insert(newOAuth);

            log.info("OAuth2账号绑定成功: userId={}, provider={}", existingUser.getId(), userInfo.getProvider());
            return OAuth2LoginResult.newBinding(existingUser);
        }

        // 3. 创建新用户
        User newUser = createUserFromOAuth(userInfo);
        UserOAuth newOAuth = createOAuthBinding(newUser.getId(), userInfo);
        userOAuthMapper.insert(newOAuth);

        log.info("OAuth2新用户注册成功: userId={}, provider={}", newUser.getId(), userInfo.getProvider());
        return OAuth2LoginResult.newUser(newUser);
    }

    private void updateOAuthLoginInfo(UserOAuth oauth, OAuth2UserInfo userInfo) {
        oauth.setLastLoginTime(LocalDateTime.now());
        oauth.setLoginCount(oauth.getLoginCount() + 1);

        // 更新用户信息（如果有变化）
        if (!Objects.equals(oauth.getName(), userInfo.getName())) {
            oauth.setName(userInfo.getName());
        }
        if (!Objects.equals(oauth.getAvatar(), userInfo.getAvatar())) {
            oauth.setAvatar(userInfo.getAvatar());
        }

        userOAuthMapper.updateById(oauth);
    }

    private User createUserFromOAuth(OAuth2UserInfo userInfo) {
        User user = new User();
        user.setUsername(generateUniqueUsername(userInfo.getName()));
        user.setEmail(userInfo.getEmail());
        user.setPassword(passwordEncoder.encode(UUID.randomUUID().toString())); // 随机密码
        user.setAvatar(userInfo.getAvatar());
        user.setRole("user");
        user.setStatus(1);

        userService.save(user);

        // 为新用户创建账户信息
        userService.createUserAccount(user.getId());

        return user;
    }

    private UserOAuth createOAuthBinding(Long userId, OAuth2UserInfo userInfo) {
        UserOAuth oauth = new UserOAuth();
        oauth.setUserId(userId);
        oauth.setProvider(userInfo.getProvider());
        oauth.setProviderId(userInfo.getId());
        oauth.setEmail(userInfo.getEmail());
        oauth.setName(userInfo.getName());
        oauth.setAvatar(userInfo.getAvatar());
        oauth.setLoginCount(1);
        oauth.setLastLoginTime(LocalDateTime.now());
        oauth.setStatus(1);

        // 加密存储令牌（如果有）
        if (userInfo.getAccessToken() != null) {
            oauth.setAccessToken(securityUtil.encryptToken(userInfo.getAccessToken()));
        }
        if (userInfo.getRefreshToken() != null) {
            oauth.setRefreshToken(securityUtil.encryptToken(userInfo.getRefreshToken()));
        }
        if (userInfo.getExpiresAt() != null) {
            oauth.setExpiresAt(userInfo.getExpiresAt());
        }

        return oauth;
    }

    private String generateUniqueUsername(String baseName) {
        if (baseName == null || baseName.trim().isEmpty()) {
            baseName = "user";
        }

        // 清理用户名，只保留字母数字和下划线
        String cleanName = baseName.replaceAll("[^a-zA-Z0-9_]", "");
        if (cleanName.isEmpty()) {
            cleanName = "user";
        }

        // 检查用户名是否已存在
        String username = cleanName;
        int counter = 1;
        while (userService.getUserByUsername(username) != null) {
            username = cleanName + "_" + counter++;
        }

        return username;
    }

    @Override
    public String buildErrorRedirectUrl(String errorCode, String errorMessage) {
        try {
            String encodedMessage = URLEncoder.encode(errorMessage, StandardCharsets.UTF_8);
            return errorRedirectUrl + "?error=" + errorCode + "&message=" + encodedMessage;
        } catch (Exception e) {
            log.error("构建错误重定向URL失败", e);
            return errorRedirectUrl + "?error=unknown";
        }
    }

    @Override
    public String getSuccessRedirectUrl() {
        return successRedirectUrl;
    }
}
```
