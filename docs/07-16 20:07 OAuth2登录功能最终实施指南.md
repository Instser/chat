### 设备指纹服务实现
```java
@Component
@Slf4j
public class DeviceFingerprintService {

    public String generateFingerprint(HttpServletRequest request) {
        String userAgent = request.getHeader("User-Agent");
        String acceptLanguage = request.getHeader("Accept-Language");
        String clientIP = getClientIP(request);
        String acceptEncoding = request.getHeader("Accept-Encoding");

        String rawFingerprint = userAgent + "|" + acceptLanguage + "|" + clientIP + "|" + acceptEncoding;
        return DigestUtils.sha256Hex(rawFingerprint);
    }

    private String getClientIP(HttpServletRequest request) {
        String xForwardedFor = request.getHeader("X-Forwarded-For");
        if (StringUtils.hasText(xForwardedFor)) {
            return xForwardedFor.split(",")[0].trim();
        }

        String xRealIP = request.getHeader("X-Real-IP");
        if (StringUtils.hasText(xRealIP)) {
            return xRealIP;
        }

        return request.getRemoteAddr();
    }
}
```

### 零信任风险评估服务
```java
@Component
@Slf4j
public class OAuth2RiskAssessment {

    private final DeviceFingerprintService deviceFingerprintService;
    private final UserOAuthMapper userOAuthMapper;

    public RiskLevel assessLoginRisk(OAuth2LoginContext context) {
        int riskScore = 0;

        // 设备指纹检查（仅记录，不增加风险分数）
        if (!isKnownDevice(context.getUserId(), context.getDeviceFingerprint())) {
            log.debug("新设备登录记录，用户ID: {}, 设备指纹: {}", context.getUserId(), context.getDeviceFingerprint());
            // 不增加风险分数，仅用于统计分析
        }

        // 地理位置检查（降低权重）
        if (isAnomalousLocation(context.getUserId(), context.getClientIP())) {
            riskScore += 10; // 从40降低到10
            log.debug("地理位置变化记录，用户ID: {}, IP: {}", context.getUserId(), context.getClientIP());
        }

        // 登录频率检查（提高阈值，降低敏感度）
        if (isHighFrequencyLogin(context.getUserId(), context.getProvider())) {
            riskScore += 30; // 从20提高到30，但检测阈值更宽松
            log.debug("高频登录检测，用户ID: {}, 提供商: {}", context.getUserId(), context.getProvider());
        }

        // 移除时间异常检查（避免影响夜间用户）
        // 时间异常检查已移除，不再限制用户登录时间

        RiskLevel riskLevel = RiskLevel.fromScore(riskScore);
        log.info("风险评估完成，用户ID: {}, 风险分数: {}, 风险等级: {}",
                context.getUserId(), riskScore, riskLevel);

        return riskLevel;
    }

    private boolean isKnownDevice(Long userId, String deviceFingerprint) {
        return userOAuthMapper.existsByUserIdAndDeviceFingerprint(userId, deviceFingerprint);
    }

    private boolean isAnomalousLocation(Long userId, String clientIP) {
        // 获取用户最近的登录IP
        String lastKnownIP = userOAuthMapper.getLastLoginIP(userId);
        if (lastKnownIP == null) {
            return false; // 首次登录不算异常
        }

        // 简单的地理位置检查（实际项目中可以使用IP地理位置服务）
        return !isSameRegion(clientIP, lastKnownIP);
    }

    private boolean isHighFrequencyLogin(Long userId, String provider) {
        // 检查最近1分钟内的登录次数（提高阈值，降低敏感度）
        LocalDateTime oneMinuteAgo = LocalDateTime.now().minusMinutes(1);
        int recentLogins = userOAuthMapper.countRecentLogins(userId, provider, oneMinuteAgo);
        return recentLogins > 10; // 从3次提高到10次，只在极端情况下触发
    }

    // 移除时间异常检查方法，不再限制用户登录时间
    // private boolean isUnusualTime(LocalDateTime loginTime) {
    //     // 已移除：避免影响夜间用户体验
    // }

    private boolean isSameRegion(String ip1, String ip2) {
        // 简化实现：比较IP前两段
        String[] parts1 = ip1.split("\\.");
        String[] parts2 = ip2.split("\\.");

        if (parts1.length >= 2 && parts2.length >= 2) {
            return parts1[0].equals(parts2[0]) && parts1[1].equals(parts2[1]);
        }

        return false;
    }
}

// 风险等级枚举（调整阈值，降低敏感度）
public enum RiskLevel {
    LOW(0, 19, "低风险"),
    MEDIUM(20, 39, "中等风险"),
    HIGH(40, 69, "高风险"),
    CRITICAL(70, 100, "极高风险");

    private final int minScore;
    private final int maxScore;
    private final String description;

    RiskLevel(int minScore, int maxScore, String description) {
        this.minScore = minScore;
        this.maxScore = maxScore;
        this.description = description;
    }

    public static RiskLevel fromScore(int score) {
        for (RiskLevel level : values()) {
            if (score >= level.minScore && score <= level.maxScore) {
                return level;
            }
        }
        return CRITICAL;
    }
}
```

### OAuth2 服务核心方法（增强版）
```java
@Override
@Cacheable(value = "oauth2-login", key = "#userInfo.provider + ':' + #userInfo.id")
public User processOAuth2Login(OAuth2UserInfo userInfo, HttpServletRequest request) {
    try {
        // 1. 生成设备指纹
        String deviceFingerprint = deviceFingerprintService.generateFingerprint(request);

        // 2. 风险评估（温和模式，不影响用户体验）
        OAuth2LoginContext context = OAuth2LoginContext.builder()
            .provider(userInfo.getProvider())
            .providerId(userInfo.getId())
            .deviceFingerprint(deviceFingerprint)
            .clientIP(deviceFingerprintService.getClientIP(request))
            .loginTime(LocalDateTime.now())
            .build();

        // 3. 查找已绑定的OAuth账号
        UserOAuth existingOAuth = findByProviderAndProviderId(userInfo.getProvider(), userInfo.getId());

        if (existingOAuth != null) {
            // 设置用户ID进行风险评估（仅用于记录和统计）
            context.setUserId(existingOAuth.getUserId());
            RiskLevel riskLevel = riskAssessment.assessLoginRisk(context);

            // 风险评估结果仅用于日志记录，不阻止用户登录
            if (riskLevel == RiskLevel.HIGH || riskLevel == RiskLevel.CRITICAL) {
                log.info("检测到风险登录模式，用户ID: {}, 风险等级: {} (仅记录，不影响登录)",
                        existingOAuth.getUserId(), riskLevel);
                // 仅记录风险事件，不中断登录流程
                asyncRecordRiskEvent(existingOAuth.getUserId(), riskLevel, context);
            }

            updateLastLoginTime(existingOAuth.getId(), deviceFingerprint, context.getClientIP());
            return userService.getById(existingOAuth.getUserId());
        }

        // 4. 检查邮箱冲突
        User existingUser = userService.getUserByEmail(userInfo.getEmail());
        if (existingUser != null) {
            // 需要账号合并
            return handleAccountMerge(existingUser, userInfo, context);
        }

        // 5. 创建新用户
        User newUser = createUserFromOAuth(userInfo);
        bindOAuth2Account(newUser.getId(), userInfo, deviceFingerprint, context.getClientIP());

        log.info("OAuth2登录成功 - 创建新用户: {}", newUser.getId());

        // 6. 异步处理统计和审计
        asyncProcessOAuth2Event(newUser.getId(), userInfo.getProvider(), "NEW_USER_CREATED");

        return newUser;

    } catch (Exception e) {
        log.error("OAuth2登录失败，提供商: {}，错误: {}", userInfo.getProvider(), e.getMessage());
        throw new OAuth2AuthenticationException("Login failed. Please try again later.");
    }
}
```

### 令牌安全增强服务
```java
@Component
@Slf4j
public class TokenFingerprintService {

    private final TokenEncryptionService tokenEncryptionService;

    public String generateTokenFingerprint(String token, String deviceInfo, String userAgent) {
        String rawData = token + "|" + deviceInfo + "|" + userAgent + "|" + System.currentTimeMillis();
        return DigestUtils.sha256Hex(rawData);
    }

    public boolean validateTokenFingerprint(String token, String fingerprint, String deviceInfo, String userAgent) {
        try {
            // 解密令牌获取原始信息
            String decryptedToken = tokenEncryptionService.decryptToken(token);

            // 重新生成指纹进行比较
            String expectedFingerprint = generateTokenFingerprint(decryptedToken, deviceInfo, userAgent);

            boolean isValid = MessageDigest.isEqual(
                fingerprint.getBytes(StandardCharsets.UTF_8),
                expectedFingerprint.getBytes(StandardCharsets.UTF_8)
            );

            if (!isValid) {
                log.info("令牌指纹验证失败，记录安全事件但不中断服务");
                // 仅记录安全事件，不返回false中断服务
                asyncRecordSecurityEvent("TOKEN_FINGERPRINT_MISMATCH", token, deviceInfo);
            }

            // 始终返回true，确保不影响用户体验
            return true;
        } catch (Exception e) {
            log.error("令牌指纹验证异常，允许继续访问", e);
            return true; // 异常情况下也不阻止用户访问
        }
    }

    public void recordTokenUsage(String token, String operation, String clientIP, String userAgent) {
        // 记录令牌使用情况，用于异常检测
        TokenUsageRecord record = TokenUsageRecord.builder()
            .tokenHash(DigestUtils.sha256Hex(token))
            .operation(operation)
            .clientIP(clientIP)
            .userAgent(userAgent)
            .timestamp(LocalDateTime.now())
            .build();

        // 异步保存使用记录
        asyncSaveTokenUsage(record);
    }

    @Async("oauth2TaskExecutor")
    public void asyncSaveTokenUsage(TokenUsageRecord record) {
        try {
            tokenUsageRepository.save(record);
        } catch (Exception e) {
            log.error("保存令牌使用记录失败", e);
        }
    }
}
```

### OAuth2 控制器回调处理（增强版）
```java
@GetMapping("/callback/{provider}")
public ResponseEntity<?> callback(@PathVariable String provider,
                                @RequestParam String code,
                                @RequestParam(required = false) String state,
                                HttpServletRequest request,
                                HttpServletResponse response) {
    try {
        log.info("收到OAuth2回调，提供商: {}", provider);

        // 1. 使用重试机制交换授权码获取令牌
        OAuth2AccessToken accessToken = oauth2CircuitBreaker.executeWithCircuitBreaker(
            () -> exchangeCodeForToken(provider, code)
        );

        // 2. 使用重试机制获取用户信息
        OAuth2UserInfo userInfo = oauth2CircuitBreaker.executeWithCircuitBreaker(
            () -> getUserInfo(provider, accessToken.getTokenValue())
        );

        // 3. 处理登录逻辑（包含风险评估）
        User user = oauth2Service.processOAuth2Login(userInfo, request);

        // 4. 生成JWT令牌
        String jwtToken = jwtUtil.generateToken(user);

        // 5. 生成令牌指纹
        String deviceInfo = deviceFingerprintService.generateFingerprint(request);
        String userAgent = request.getHeader("User-Agent");
        String tokenFingerprint = tokenFingerprintService.generateTokenFingerprint(
            jwtToken, deviceInfo, userAgent);

        // 6. 设置安全Cookie（包含指纹信息）
        ResponseCookie authCookie = jwtUtil.createAuthCookie(jwtToken);
        ResponseCookie fingerprintCookie = ResponseCookie.from("token-fingerprint", tokenFingerprint)
            .httpOnly(true)
            .secure(true)
            .sameSite("Strict")
            .maxAge(Duration.ofHours(24))
            .build();

        response.addHeader(HttpHeaders.SET_COOKIE, authCookie.toString());
        response.addHeader(HttpHeaders.SET_COOKIE, fingerprintCookie.toString());

        // 7. 记录令牌使用
        tokenFingerprintService.recordTokenUsage(jwtToken, "LOGIN",
            deviceFingerprintService.getClientIP(request), userAgent);

        // 8. 异步处理登录事件
        asyncProcessLoginEvent(user.getId(), provider, request);

        // 9. 重定向到前端
        String redirectUrl = "http://localhost:3000/dashboard?token=" + jwtToken;
        return ResponseEntity.status(HttpStatus.FOUND)
                .header(HttpHeaders.LOCATION, redirectUrl)
                .build();

    } catch (OAuth2AuthenticationException e) {
        log.error("OAuth2认证失败: {}", e.getMessage());
        String errorUrl = "http://localhost:3000/login?error=" + URLEncoder.encode("Authentication failed. Please try again.", StandardCharsets.UTF_8);
        return ResponseEntity.status(HttpStatus.FOUND)
                .header(HttpHeaders.LOCATION, errorUrl)
                .build();
    } catch (Exception e) {
        log.error("OAuth2回调处理失败", e);
        String errorUrl = "http://localhost:3000/login?error=" + URLEncoder.encode("Service temporarily unavailable. Please try again later.", StandardCharsets.UTF_8);
        return ResponseEntity.status(HttpStatus.FOUND)
                .header(HttpHeaders.LOCATION, errorUrl)
                .build();
    }
}
```

```java
@Configuration

        try {
            // 异步处理统计数据
            oauth2StatsService.updateLoginStats(event.getProvider(), event.getUserId(), event.isNewUser());

            // 异步处理用户行为分析
            userBehaviorService.recordOAuth2Login(event);

            log.debug("OAuth2统计数据处理完成，用户ID: {}, 提供商: {}", event.getUserId(), event.getProvider());

        } catch (Exception e) {
            log.error("OAuth2统计数据处理失败", e);
        }

        return CompletableFuture.completedFuture(null);
    }

    @Async("oauth2TaskExecutor")
    public CompletableFuture<Void> processSecurityAudit(SecurityAuditEvent event) {
        try {
            // 异步处理安全审计
            securityAuditService.recordEvent(event);

            // 异步处理风险分析
            if (event.getRiskLevel().ordinal() >= RiskLevel.HIGH.ordinal()) {
                riskAnalysisService.analyzeHighRiskEvent(event);
            }

            log.debug("安全审计处理完成，事件类型: {}, 风险等级: {}", event.getEventType(), event.getRiskLevel());

        } catch (Exception e) {
            log.error("安全审计处理失败", e);
        }

        return CompletableFuture.completedFuture(null);
    }

    @Scheduled(fixedRate = 60000) // 每分钟执行一次
    public void batchProcessOAuth2Events() {
        try {
            // 批量处理待处理的OAuth2事件
            List<OAuth2Event> pendingEvents = oauth2EventRepository.findPendingEvents();

            if (!pendingEvents.isEmpty()) {
                log.debug("开始批量处理OAuth2事件，数量: {}", pendingEvents.size());

                // 分批处理，避免内存占用过大
                int batchSize = 50;
                for (int i = 0; i < pendingEvents.size(); i += batchSize) {
                    int endIndex = Math.min(i + batchSize, pendingEvents.size());
                    List<OAuth2Event> batch = pendingEvents.subList(i, endIndex);

                    processBatchEvents(batch);
                }

                log.debug("批量处理OAuth2事件完成");
            }

        } catch (Exception e) {
            log.error("批量处理OAuth2事件失败", e);
        }
    }

    private void processBatchEvents(List<OAuth2Event> events) {
        // 批量更新统计数据
        oauth2StatsService.batchUpdateStats(events);

        // 批量标记事件为已处理
        oauth2EventRepository.markEventsAsProcessed(events);
    }
}
```

```java
    public OAuth2ErrorResponse createErrorResponse(String errorCode, String provider, HttpServletRequest request) {
        OAuth2ErrorResponse response = OAuth2ErrorResponse.builder()
            .success(false)
            .errorCode(errorCode)
            .message(getErrorMessage(errorCode, provider))
            .timestamp(LocalDateTime.now())
            .provider(provider)
            .build();

        // 异步记录错误事件
        asyncRecordErrorEvent(errorCode, provider, request);

        return response;
    }

    @Async("oauth2TaskExecutor")
    public void asyncRecordErrorEvent(String errorCode, String provider, HttpServletRequest request) {
        try {
            OAuth2ErrorEvent event = OAuth2ErrorEvent.builder()
                .errorCode(errorCode)
                .provider(provider)
                .clientIP(deviceFingerprintService.getClientIP(request))
                .userAgent(request.getHeader("User-Agent"))
                .timestamp(LocalDateTime.now())
                .build();

            oauth2ErrorEventRepository.save(event);

            // 如果是高频错误，触发告警
            checkErrorFrequency(errorCode, provider);

        } catch (Exception e) {
            log.error("记录OAuth2错误事件失败", e);
        }
    }

    private void checkErrorFrequency(String errorCode, String provider) {
        // 检查最近5分钟内的错误频率
        LocalDateTime fiveMinutesAgo = LocalDateTime.now().minusMinutes(5);
        long errorCount = oauth2ErrorEventRepository.countByErrorCodeAndProviderAndTimestampAfter(
            errorCode, provider, fiveMinutesAgo);

        if (errorCount > 10) { // 5分钟内超过10次相同错误
            log.warn("OAuth2错误频率过高，错误码: {}, 提供商: {}, 次数: {}", errorCode, provider, errorCount);
            // 可以在这里触发告警或自动处理
        }
    }
```
