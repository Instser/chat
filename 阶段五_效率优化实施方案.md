### 3.1 第一步：实现请求级权限缓存（2天）

#### 3.1.1 请求级缓存管理器
```java
@Component
public class RequestLevelPermissionCache {

    private static final ThreadLocal<Map<String, Object>> REQUEST_CACHE = 
        ThreadLocal.withInitial(HashMap::new);

    /**
     * 获取或计算缓存值
     */
    public <T> T getOrCompute(String key, Supplier<T> supplier) {
        Map<String, Object> cache = REQUEST_CACHE.get();
        
        @SuppressWarnings("unchecked")
        T result = (T) cache.get(key);
        
        if (result == null) {
            result = supplier.get();
            if (result != null) {
                cache.put(key, result);
            }
        }
        
        return result;
    }

    /**
     * 直接存储值到缓存
     */
    public void put(String key, Object value) {
        if (value != null) {
            REQUEST_CACHE.get().put(key, value);
        }
    }

    /**
     * 获取缓存值
     */
    public <T> T get(String key) {
        @SuppressWarnings("unchecked")
        T result = (T) REQUEST_CACHE.get().get(key);
        return result;
    }

    /**
     * 清除请求级缓存
     */
    public static void clear() {
        REQUEST_CACHE.remove();
    }

    /**
     * 获取缓存统计信息
     */
    public Map<String, Object> getCacheStats() {
        Map<String, Object> cache = REQUEST_CACHE.get();
        Map<String, Object> stats = new HashMap<>();
        stats.put("size", cache.size());
        stats.put("keys", new ArrayList<>(cache.keySet()));
        return stats;
    }
}
```

```java
    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain)
            throws IOException, ServletException {
        try {
            chain.doFilter(request, response);
        } finally {
            // 确保每次请求结束后清理缓存
            RequestLevelPermissionCache.clear();
        }
    }

    @Override
    public void init(FilterConfig filterConfig) {
        // 过滤器初始化
    }

    }
```

### 3.2 第二步：优化权限验证链（2天）

#### 3.2.1 高效权限验证器
```java
@Component
@Slf4j
public class OptimizedPermissionValidator {
    
    @Autowired
    private RequestLevelPermissionCache requestCache;
    
    @Autowired
    private UserAccountService userAccountService;
    
    @Autowired
    private ChatLimitService chatLimitService;
    
    @Autowired
    private ChatInterruptService chatInterruptService;

    /**
     * 优化的统一权限验证
     */
    public PermissionValidationResult validateAllPermissions(
            Long userId, String sessionId, Long modelId, Authentication authentication) {
        
        // 使用请求级缓存避免重复验证
        String cacheKey = String.format("permission:%d:%s:%d", userId, sessionId, modelId);
        
        return requestCache.getOrCompute(cacheKey, () -> {
            return performOptimizedValidation(userId, sessionId, modelId, authentication);
        });
    }

    /**
     * 执行优化的权限验证
     */
    private PermissionValidationResult performOptimizedValidation(
            Long userId, String sessionId, Long modelId, Authentication authentication) {
        
        try {
            // 按验证成本排序，先执行轻量级验证
            
            // 1. 用户认证验证（最轻量）
            PermissionValidationResult authResult = validateUserAuthenticationCached(authentication, userId);
            if (authResult.isFailure()) {
                return authResult;
            }

            // 2. 活跃会话检查（轻量）
            PermissionValidationResult sessionResult = validateActiveSessionCached(userId);
            if (sessionResult.isFailure()) {
                return sessionResult;
            }

            // 3. 用户账户状态验证（中等）
            PermissionValidationResult accountResult = validateUserAccountCached(userId);
            if (accountResult.isFailure()) {
                return accountResult;
            }

            // 4. 聊天权限验证（重量）
            PermissionValidationResult chatResult = validateChatPermissionsCached(userId, sessionId, modelId);
            if (chatResult.isFailure()) {
                return chatResult;
            }

            // 所有验证通过
            return PermissionValidationResult.success(
                accountResult.getUserAccount(), 
                accountResult.getPackageInfo()
            );

        } catch (Exception e) {
            LogUtils.logUserError(log, userId, "权限验证", "优化权限验证异常", e);
            return PermissionValidationResult.failure("SYSTEM_ERROR", "系统异常，请稍后重试");
        }
    }

    /**
     * 缓存用户认证验证
     */
    private PermissionValidationResult validateUserAuthenticationCached(Authentication authentication, Long userId) {
        String cacheKey = "auth:" + userId;
        return requestCache.getOrCompute(cacheKey, () -> {
            return validateUserAuthentication(authentication, userId);
        });
    }

    /**
     * 缓存活跃会话检查
     */
    private PermissionValidationResult validateActiveSessionCached(Long userId) {
        String cacheKey = "active_session:" + userId;
        return requestCache.getOrCompute(cacheKey, () -> {
            if (chatInterruptService.hasActiveSession(userId)) {
                return PermissionValidationResult.failure("ACTIVE_SESSION_EXISTS", 
                    "当前正在聊天中，请结束当前聊天后重试");
            }
            return PermissionValidationResult.success(null, null);
        });
    }

    /**
     * 缓存用户账户验证
     */
    private PermissionValidationResult validateUserAccountCached(Long userId) {
        String cacheKey = "account:" + userId;
        return requestCache.getOrCompute(cacheKey, () -> {
            UserAccount userAccount = userAccountService.getUserAccount(userId);
            if (userAccount == null) {
                return PermissionValidationResult.failure("ACCOUNT_NOT_FOUND", "用户账户不存在");
            }

            if (userAccount.isPackageExpired()) {
                return PermissionValidationResult.failure("PACKAGE_EXPIRED", "套餐已过期");
            }

            return PermissionValidationResult.success(userAccount, null);
        });
    }

    /**
     * 缓存聊天权限验证
     */
    private PermissionValidationResult validateChatPermissionsCached(Long userId, String sessionId, Long modelId) {
        String cacheKey = "chat_permission:" + userId + ":" + modelId;
        return requestCache.getOrCompute(cacheKey, () -> {
            String permissionError = chatLimitService.validateChatPermission(userId, sessionId, modelId);
            if (permissionError != null) {
                String errorMessage = getErrorMessage(permissionError);
                return PermissionValidationResult.failure(permissionError, errorMessage);
            }
            return PermissionValidationResult.success(null, null);
        });
    }

    // 其他辅助方法...
}
```

### 3.3 第三步：智能权限预加载（2天）

#### 3.3.1 权限预加载管理器
```java
@Component
@Slf4j
public class PermissionPreloadManager {
    
    @Autowired
    private UserAccountService userAccountService;
    
    @Autowired
    private HybridCacheManager hybridCacheManager;
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;

    /**
     * 预加载用户权限信息
     */
    @Async("permissionPreloadExecutor")
    public CompletableFuture<Void> preloadUserPermissions(Long userId) {
        try {
            // 1. 预加载用户账户信息
            preloadUserAccount(userId);
            
            // 2. 预加载用户套餐信息
            preloadUserPackage(userId);
            
            // 3. 预加载常用模型权限
            preloadCommonModelPermissions(userId);
            
            LogUtils.logUserDebug(log, userId, "权限预加载", "用户权限信息预加载完成");
            
        } catch (Exception e) {
            LogUtils.logUserError(log, userId, "权限预加载", "权限预加载异常", e);
        }
        
        return CompletableFuture.completedFuture(null);
    }

    /**
     * 预加载用户账户信息
     */
    private void preloadUserAccount(Long userId) {
        String cacheKey = "user:account:" + userId;
        
        // 检查缓存是否已存在
        if (hybridCacheManager.exists(cacheKey)) {
            return;
        }
        
        // 异步加载并缓存
        UserAccount account = userAccountService.getUserAccountFromDatabase(userId);
        if (account != null) {
            hybridCacheManager.put(cacheKey, account, Duration.ofMinutes(15));
        }
    }

    /**
     * 预加载用户套餐信息
     */
    private void preloadUserPackage(Long userId) {
        try {
            UserAccount account = userAccountService.getUserAccount(userId);
            if (account != null && account.getPackageId() != null) {
                String packageCacheKey = "package:config:" + account.getPackageId();
                
                if (!hybridCacheManager.exists(packageCacheKey)) {
                    Package packageInfo = packageService.getPackageById(account.getPackageId());
                    if (packageInfo != null) {
                        hybridCacheManager.put(packageCacheKey, packageInfo, Duration.ofHours(1));
                    }
                }
            }
        } catch (Exception e) {
            LogUtils.logUserError(log, userId, "权限预加载", "预加载套餐信息异常", e);
        }
    }

    /**
     * 预加载常用模型权限
     */
    private void preloadCommonModelPermissions(Long userId) {
        try {
            // 获取用户常用的模型ID列表
            List<Long> commonModelIds = getUserCommonModels(userId);
            
            for (Long modelId : commonModelIds) {
                String permissionCacheKey = "model:permission:" + userId + ":" + modelId;
                
                if (!hybridCacheManager.exists(permissionCacheKey)) {
                    // 预计算模型权限
                    String permissionResult = chatLimitService.validateModelPermissionOnly(userId, modelId);
                    hybridCacheManager.put(permissionCacheKey, permissionResult, Duration.ofMinutes(30));
                }
            }
        } catch (Exception e) {
            LogUtils.logUserError(log, userId, "权限预加载", "预加载模型权限异常", e);
        }
    }

    /**
     * 获取用户常用模型
     */
    private List<Long> getUserCommonModels(Long userId) {
        // 从Redis获取用户最近使用的模型
        String key = "user:recent_models:" + userId;
        
        try {
            @SuppressWarnings("unchecked")
            List<Long> recentModels = (List<Long>) redisTemplate.opsForValue().get(key);
            
            if (recentModels != null && !recentModels.isEmpty()) {
                return recentModels.subList(0, Math.min(5, recentModels.size()));
            }
        } catch (Exception e) {
            LogUtils.logUserWarning(log, userId, "权限预加载", "获取常用模型失败", e);
        }
        
        // 返回默认常用模型
        return Arrays.asList(1L, 2L, 3L);
    }

    /**
     * 用户登录时触发权限预加载
     */
    @EventListener
    public void handleUserLoginEvent(UserLoginEvent event) {
        Long userId = event.getUserId();
        LogUtils.logUserInfo(log, userId, "权限预加载", "用户登录，开始预加载权限信息");
        preloadUserPermissions(userId);
    }
}
```

```java
@Component
@Slf4j
public class PermissionPerformanceMonitor {
    
    private final MeterRegistry meterRegistry;
    private final Timer permissionValidationTimer;
    private final Counter permissionCacheHitCounter;
    private final Counter permissionCacheMissCounter;
    
    public PermissionPerformanceMonitor(MeterRegistry meterRegistry) {
        this.meterRegistry = meterRegistry;
        this.permissionValidationTimer = Timer.builder("permission.validation.time")
            .description("Permission validation time")
            .register(meterRegistry);
        this.permissionCacheHitCounter = Counter.builder("permission.cache.hit")
            .description("Permission cache hit count")
            .register(meterRegistry);
        this.permissionCacheMissCounter = Counter.builder("permission.cache.miss")
            .description("Permission cache miss count")
            .register(meterRegistry);
    }
    
    /**
     * 记录权限验证时间
     */
    public <T> T timePermissionValidation(String operation, Supplier<T> supplier) {
        return permissionValidationTimer.recordCallable(() -> {
            return supplier.get();
        });
    }
    
    /**
     * 记录缓存命中
     */
    public void recordCacheHit(String cacheType) {
        permissionCacheHitCounter.increment(Tags.of("type", cacheType));
    }
    
    /**
     * 记录缓存未命中
     */
    public void recordCacheMiss(String cacheType) {
        permissionCacheMissCounter.increment(Tags.of("type", cacheType));
    }
    
    /**
     * 获取权限验证性能报告
     */
    public Map<String, Object> getPerformanceReport() {
        Map<String, Object> report = new HashMap<>();
        
        // 平均验证时间
        report.put("avgValidationTime", permissionValidationTimer.mean(TimeUnit.MILLISECONDS));
        
        // 缓存命中率
        double hitCount = permissionCacheHitCounter.count();
        double missCount = permissionCacheMissCounter.count();
        double hitRate = hitCount / (hitCount + missCount);
        report.put("cacheHitRate", hitRate);
        
        // 总验证次数
        report.put("totalValidations", permissionValidationTimer.count());
        
        return report;
    }
}
```
